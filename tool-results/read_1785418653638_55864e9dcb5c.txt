   488→   488→            const inCategory = e.category.toLowerCase().includes(query);
   489→   489→            const inId = e.id.toLowerCase().includes(query);
   490→   490→            return inName || inDesc || inTags || inCategory || inId;

   494→   494→        const sliced = results.slice(0, limit);

    27→/* ═══════════════════════════════════════════════════════════════
    28→   COLOR MATH — OKLCH (perceptually uniform color space)
    29→   Reference: Björn Ottosson, "A perceptual color space for image
    30→   processing" (Dec 2020). https://bottosson.github.io/posts/oklab/
    31→   All channel values are in [0,1] unless otherwise noted.
    32→
    33→   Sanity checks (verified by hand):
    34→     #ffffff → L=1.000  C=0  H=0
    35→     #000000 → L=0.000  C=0  H=0
    36→     #808080 → L≈0.596  C≈0  H=0
    37→   ═══════════════════════════════════════════════════════════════ */
    38→
    39→/** sRGB [0,1] → linear sRGB [0,1] (inverse gamma). */
    40→function srgbToLinear(c: number): number {
    41→  return c <= 0.04045 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4);
    42→}
    43→
    44→/** Linear sRGB [0,1] → sRGB [0,1] (gamma encode). */
    45→function linearToSrgb(c: number): number {
    46→  return c <= 0.0031308 ? c * 12.92 : 1.055 * Math.pow(c, 1 / 2.4) - 0.055;
    47→}
    48→
    49→/** Linear sRGB → OKLab. Returns [L, a, b]. */
    50→function linearSrgbToOklab(r: number, g: number, b: number): [number, number, number] {
    51→  const l = 0.4122214708 * r + 0.5363325363 * g + 0.0514459929 * b;
    52→  const m = 0.2119034982 * r + 0.6806995451 * g + 0.1073969566 * b;
    53→  const s = 0.0883024619 * r + 0.2817188376 * g + 0.6299787005 * b;
    54→
    55→  const l_ = Math.cbrt(l);
    56→  const m_ = Math.cbrt(m);
    57→  const s_ = Math.cbrt(s);
    58→
    59→  return [
    60→    0.2104542553 * l_ + 0.7936177850 * m_ - 0.0040720468 * s_,
    61→    1.9779984951 * l_ - 2.4285922050 * m_ + 0.4505937099 * s_,
    62→    0.0259040371 * l_ + 0.7827717662 * m_ - 0.8086757660 * s_,
    63→  ];
    64→}
    65→
    66→/** OKLab → linear sRGB. Returns [r, g, b]. */
    67→function oklabToLinearSrgb(L: number, a: number, b: number): [number, number, number] {
    68→  const l_ = L + 0.3963377774 * a + 0.2158037573 * b;
    69→  const m_ = L - 0.1055613458 * a - 0.0638541728 * b;
    70→  const s_ = L - 0.0894841775 * a - 1.2914855480 * b;
    71→
    72→  const l = l_ * l_ * l_;
    73→  const m = m_ * m_ * m_;
    74→  const s = s_ * s_ * s_;
    75→
    76→  return [
    77→    4.0767416621 * l - 3.3077115913 * m + 0.2309699292 * s,
    78→    -1.2684380046 * l + 2.6097574011 * m - 0.3413193965 * s,
    79→    -0.0041960863 * l - 0.7034186147 * m + 1.7076147010 * s,
    80→  ];
    81→}

    83→interface Oklch {
    84→  /** Perceptual lightness, 0..1 (0=black, 1=white). */
    85→  L: number;
    86→  /** Chroma, 0..~0.4. */
    87→  C: number;
    88→  /** Hue angle, 0..360 degrees. */
    89→  H: number;
    90→}
    91→
    92→/** OKLab → OKLCH (H in degrees, 0..360). */
    93→function oklabToOklch(L: number, a: number, b: number): Oklch {
    94→  const C = Math.sqrt(a * a + b * b);
    95→  let H = (Math.atan2(b, a) * 180) / Math.PI;
    96→  if (H < 0) H += 360;
    97→  return { L, C, H };
    98→}
    99→
   100→/** OKLCH → OKLab (returns [L, a, b]). */
   101→function oklchToOklab({ L, C, H }: Oklch): [number, number, number] {
   102→  const hRad = (H * Math.PI) / 180;
   103→  return [L, C * Math.cos(hRad), C * Math.sin(hRad)];
   104→}
   105→
   106→/** Hex string (#rgb | #rgba | #rrggbb | #rrggbbaa) → OKLCH, or null if invalid. */
   107→function hexToOklch(hex: string): Oklch | null {
   108→  const rgb = parseHexToRgb01(hex);
   109→  if (!rgb) return null;
   110→  const [r, g, b] = rgb;
   111→  const [L, a, b_] = linearSrgbToOklab(
   112→    srgbToLinear(r),
   113→    srgbToLinear(g),
   114→    srgbToLinear(b),
   115→  );
   116→  return oklabToOklch(L, a, b_);
   117→}
   118→
   119→/** OKLCH → #rrggbb hex string. */
   120→function oklchToHex({ L, C, H }: Oklch): string {
   121→  const [oL, oa, ob] = oklchToOklab({ L, C, H });
   122→  const [lr, lg, lb] = oklabToLinearSrgb(oL, oa, ob);
   123→  return rgbToHex(linearToSrgb(lr), linearToSrgb(lg), linearToSrgb(lb));
   124→}

   126→/** WCAG relative luminance for an sRGB triple (each in [0,1]). */
   127→function relativeLuminance(r: number, g: number, b: number): number {
   128→  return (
   129→    0.2126 * srgbToLinear(r) + 0.7152 * srgbToLinear(g) + 0.0722 * srgbToLinear(b)
   130→  );
   131→}
   132→
   133→/** WCAG contrast ratio between two sRGB triples (each in [0,1]). Returns 1..21. */
   134→function contrastRatio(
   135→  a: [number, number, number],
   136→  b: [number, number, number],
   137→): number {
   138→  const la = relativeLuminance(a[0], a[1], a[2]);
   139→  const lb = relativeLuminance(b[0], b[1], b[2]);
   140→  const [hi, lo] = la >= lb ? [la, lb] : [lb, la];
   141→  return (hi + 0.05) / (lo + 0.05);
   142→}

   148→function clamp(v: number, lo: number, hi: number): number {
   149→  return Math.max(lo, Math.min(hi, v));
   150→}

   152→function rgbToHex(r: number, g: number, b: number): string {
   153→  const toByte = (c: number) => {
   154→    const clamped = clamp(c, 0, 1);
   155→    return Math.round(clamped * 255)
   156→      .toString(16)
   157→      .padStart(2, "0");
   158→  };
   159→  return `#${toByte(r)}${toByte(g)}${toByte(b)}`;
   160→}

   162→/** Parse a hex string (#rgb, #rgba, #rrggbb, #rrggbbaa) → [r,g,b] in [0,1]. */
   163→function parseHexToRgb01(hex: string): [number, number, number] | null {
   164→  let h = hex.trim().replace(/^#/, "");
   165→  if (h.length === 3) {
   166→    h = h[0]! + h[0]! + h[1]! + h[1]! + h[2]! + h[2]!;
   167→  } else if (h.length === 4) {
   168→    // #rgba → expand + drop alpha
   169→    h = h[0]! + h[0]! + h[1]! + h[1]! + h[2]! + h[2]!;
   170→  } else if (h.length === 8) {
   171→    // #rrggbbaa → drop alpha
   172→    h = h.slice(0, 6);
   173→  }
   174→  if (!/^[0-9a-fA-F]{6}$/.test(h)) return null;
   175→  return [
   176→    parseInt(h.slice(0, 2), 16) / 255,
   177→    parseInt(h.slice(2, 4), 16) / 255,
   178→    parseInt(h.slice(4, 6), 16) / 255,
   179→  ];
   180→}
   181→
   182→/** Normalize any supported hex form to lowercase #rrggbb, or null. */
   183→function normalizeHex(hex: string): string | null {
   184→  const rgb = parseHexToRgb01(hex);
   185→  return rgb ? rgbToHex(rgb[0], rgb[1], rgb[2]) : null;
   186→}

   188→/** Convert HSL (h:0..360, s:0..1, l:0..1) → sRGB [0,1] triple. */
   189→function hslToRgb(h: number, s: number, l: number): [number, number, number] {
   190→  const c = (1 - Math.abs(2 * l - 1)) * s;
   191→  const hp = (((h % 360) + 360) % 360) / 60;
   192→  const x = c * (1 - Math.abs((hp % 2) - 1));
   193→  let r = 0;
   194→  let g = 0;
   195→  let b = 0;
   196→  if (hp >= 0 && hp < 1) {
   197→    r = c;
   198→    g = x;
   199→    b = 0;
   200→  } else if (hp < 2) {
   201→    r = x;
   202→    g = c;
   203→    b = 0;
   204→  } else if (hp < 3) {
   205→    r = 0;
   206→    g = c;
   207→    b = x;
   208→  } else if (hp < 4) {
   209→    r = 0;
   210→    g = x;
   211→    b = c;
   212→  } else if (hp < 5) {
   213→    r = x;
   214→    g = 0;
   215→    b = c;
   216→  } else {
   217→    r = c;
   218→    g = 0;
   219→    b = x;
   220→  }
   221→  const m = l - c / 2;
   222→  return [r + m, g + m, b + m];
   223→}

   225→/**
   226→ * Try to interpret a CSS color string (hex, rgb()/rgba(), hsl()/hsla())
   227→ * → #rrggbb, or null. Handles both comma and space modern syntax,
   228→ * percentages, and the `deg` hue unit.
   229→ */
   230→function anyColorToHex(input: string): string | null {
   231→  const trimmed = input.trim();
   232→  if (trimmed.startsWith("#")) return normalizeHex(trimmed);
   233→
   234→  // rgb()/rgba() — comma or space separated, with optional % and alpha
   235→  const rgbMatch = trimmed.match(
   236→    /^rgba?\(\s*([\d.]+)(%)?\s*[ ,]\s*([\d.]+)(%)?\s*[ ,]\s*([\d.]+)(%)?\s*(?:[,/]\s*[\d.]+%?\s*)?\)$/i,
   237→  );
   238→  if (rgbMatch) {
   239→    const parseChannel = (val: string | undefined, pct: string | undefined) => {
   240→      const n = parseFloat(val ?? "0");
   241→      return pct ? n / 100 : n / 255;
   242→    };
   243→    const r = parseChannel(rgbMatch[1], rgbMatch[2]);
   244→    const g = parseChannel(rgbMatch[3], rgbMatch[4]);
   245→    const b = parseChannel(rgbMatch[5], rgbMatch[6]);
   246→    return rgbToHex(r, g, b);
   247→  }
   248→
   249→  // hsl()/hsla()
   250→  const hslMatch = trimmed.match(
   251→    /^hsla?\(\s*([\d.]+)(?:deg)?\s*[ ,]\s*([\d.]+)%\s*[ ,]\s*([\d.]+)%\s*(?:[,/]\s*[\d.]+%?\s*)?\)$/i,
   252→  );
   253→  if (hslMatch) {
   254→    const h = parseFloat(hslMatch[1] ?? "0");
   255→    const s = parseFloat(hslMatch[2] ?? "0") / 100;
   256→    const l = parseFloat(hslMatch[3] ?? "0") / 100;
   257→    const [r, g, b] = hslToRgb(h, s, l);
   258→    return rgbToHex(r, g, b);
   259→  }
   260→
   261→  return null;
   262→}

   268→interface ExtractedColor {
   269→  hex: string;
   270→  label: string;
   271→}
   272→
   273→/**
   274→ * Extract colors from a CSS string. If `:root { --var: <color>; }` blocks
   275→ * are present, the variable names become labels (with `--color-`/`--c-`
   276→ * prefixes stripped). Otherwise just collect every hex/rgb()/hsl() value,
   277→ * deduped, with sequential labels.
   278→ */
   279→function extractColorsFromCss(css: string): ExtractedColor[] {
   280→  const labeled: ExtractedColor[] = [];
   281→  const seenLabeled = new Set<string>();
   282→
   283→  // Pattern 1: --var-name: <color>;
   284→  const varRe =
   285→    /--([a-zA-Z][a-zA-Z0-9-]*)\s*:\s*(#[0-9a-fA-F]{3,8}\b|rgba?\([^)]*\)|hsla?\([^)]*\))/g;
   286→  let m: RegExpExecArray | null;
   287→  while ((m = varRe.exec(css)) !== null) {
   288→    const rawLabel =
   289→      (m[1] ?? "")
   290→        .replace(/^(?:color|c)-/i, "")
   291→        .replace(/[-_]/g, " ")
   292→        .trim() || "color";
   293→    const hex = anyColorToHex(m[2] ?? "");
   294→    if (!hex) continue;
   295→    const key = `${rawLabel.toLowerCase()}:${hex}`;
   296→    if (seenLabeled.has(key)) continue;
   297→    seenLabeled.add(key);
   298→    labeled.push({ hex, label: rawLabel });
   299→  }
   300→
   301→  if (labeled.length > 0) return labeled;
   302→
   303→  // Pattern 2: any hex / rgb / hsl anywhere (deduped)
   304→  const found: string[] = [];
   305→  const seenHex = new Set<string>();
   306→
   307→  const hexRe = /#[0-9a-fA-F]{3}(?:[0-9a-fA-F]{1})?(?:[0-9a-fA-F]{2})?(?:[0-9a-fA-F]{2})?\b/g;
   308→  while ((m = hexRe.exec(css)) !== null) {
   309→    const hex = normalizeHex(m[0]);
   310→    if (hex && !seenHex.has(hex)) {
   311→      seenHex.add(hex);
   312→      found.push(hex);
   313→    }
   314→  }
   315→
   316→  const rgbRe = /rgba?\(\s*[\d.%\s,/-]+deg?\s*[\d.%\s,/-]*\)/gi;
   317→  while ((m = rgbRe.exec(css)) !== null) {
   318→    const hex = anyColorToHex(m[0]);
   319→    if (hex && !seenHex.has(hex)) {
   320→      seenHex.add(hex);
   321→      found.push(hex);
   322→    }
   323→  }
   324→
   325→  const hslRe = /hsla?\(\s*[\d.%\s,/-]+deg?\s*[\d.%\s,/-]*\)/gi;
   326→  while ((m = hslRe.exec(css)) !== null) {
   327→    const hex = anyColorToHex(m[0]);
   328→    if (hex && !seenHex.has(hex)) {
   329→      seenHex.add(hex);
   330→      found.push(hex);
   331→    }
   332→  }
   333→
   334→  return found.map((hex, i) => ({ hex, label: `Color ${i + 1}` }));
   335→}

   337→/* ═══════════════════════════════════════════════════════════════

   341→interface ConvertOptions {
   342→  hueRotation: boolean;
   343→  boostContrast: number; // 0..100
   344→}
   345→
   346→/**
   347→ * Convert a single OKLCH color to its dark-mode counterpart.
   348→ *
   349→ * Strategy (per spec):
   350→ * - Light (L > 0.65): invert L to a "dark surface" range via
   351→ *   L' = clamp(0.18, 1 - L + 0.15, 0.35). Chroma reduced ~15% on
   352→ *   very saturated colors (C > 0.10) to avoid neon glow on dark.
   353→ * - Dark (L < 0.35): invert L to a "light text" range via
   354→ *   L' = clamp(0.75, 1 - L - 0.10, 0.95). Chroma preserved.
   355→ * - Mid (0.35 ≤ L ≤ 0.65): mild invert L' = 1 - L, chroma −15%.
   356→ *
   357→ * The `boostContrast` slider (0..100) pushes L' further from the
   358→ * perceptual midpoint (0.5), increasing contrast in dark mode.
   359→ * Optional +15° hue rotation warms accents.
   360→ */
   361→function convertToDark(input: Oklch, opts: ConvertOptions): Oklch {
   362→  const { L, C, H } = input;
   363→  const boost = opts.boostContrast / 100; // 0..1
   364→
   365→  let Lp: number;
   366→  let Cp: number;
   367→
   368→  if (L > 0.65) {
   369→    // Light → dark surface
   370→    Lp = clamp(0.18, 1.0 - L + 0.15, 0.35);
   371→    // Boost: push further from 0.5 → darker
   372→    Lp = clamp(0.1, Lp - boost * 0.08, 0.35);
   373→    // Reduce chroma for very saturated colors to avoid neon glow
   374→    Cp = C > 0.1 ? C * 0.85 : C;
   375→  } else if (L < 0.35) {
   376→    // Dark → light text
   377→    Lp = clamp(0.75, 1.0 - L - 0.1, 0.95);
   378→    // Boost: push further from 0.5 → lighter
   379→    Lp = clamp(0.75, Lp + boost * 0.05, 0.98);
   380→    Cp = C;
   381→  } else {
   382→    // Mid
   383→    Lp = 1.0 - L;
   384→    // Boost: push further from 0.5
   385→    if (Lp < 0.5) {
   386→      Lp = clamp(0.1, Lp - boost * 0.08, 0.5);
   387→    } else {
   388→      Lp = clamp(0.5, Lp + boost * 0.08, 0.98);
   389→    }
   390→    Cp = C * 0.85;
   391→  }
   392→
   393→  const Hp = opts.hueRotation ? (H + 15) % 360 : H;
   394→
   395→  return { L: Lp, C: Cp, H: Hp };
   396→}

   398→/** WCAG contrast tier label for a ratio (1..21). */

   430→function slug(label: string, fallback: string): string {
   431→  const cleaned = (label || fallback)
   432→    .trim()
   433→    .toLowerCase()
   434→    .replace(/[^a-z0-9]+/g, "-")
   435→    .replace(/^-+|-+$/g, "");
   436→  return cleaned || fallback;
   437→}

   442→  );
   443→  return `:root {\n${lines.join("\n")}\n}`;
   444→}

   446→function formatTailwind(items: { label: string; hex: string }[]): string {

   451→  return `// tailwind.config.js\nmodule.exports = {\n  theme: {\n    extend: {\n      colors: {\n${lines.join(",\n")}\n      }\n    }\n  }\n}`;
   452→}

   503→  hex: string;
   504→  onChange?: (hex: string) => void;
   505→  ariaLabel: string;

   508→function ColorSwatch({ hex, onChange, ariaLabel }: SwatchProps) {
   509→  // <input type="color"> requires a 7-char #rrggbb value
   510→  const safeValue = normalizeHex(hex) ?? "#000000";
   511→  return (
   512→    <div
   513→      className={cn(
   514→        "relative size-10 shrink-0 rounded-md border border-border overflow-hidden",
   515→        onChange && "cursor-pointer",
   516→      )}
   517→      style={{ background: hex }}
   518→    >
   519→      {onChange ? (
   520→        <input
   521→          type="color"
   522→          value={safeValue}
   523→          onChange={(e) => onChange(e.target.value)}
   524→          className="absolute inset-0 size-full cursor-pointer opacity-0"
   525→          aria-label={ariaLabel}
   526→        />
   527→      ) : (
   528→        <span className="sr-only">{ariaLabel}</span>
   529→      )}
   530→    </div>
   531→  );
   532→}

   538→/**
   539→ * DarkModeConverter — takes a set of light-mode colors and auto-generates
   540→ * a perceptually-tuned dark-mode palette via OKLCH color science. Features
   541→ * palette/CSS input modes, side-by-side comparison with WCAG contrast
   542→ * indicators, a live UI preview, and CSS/Tailwind/OKLCH exports.
   543→ */
   544→export function DarkModeConverter() {
   545→  /* ─── State ─── */
   546→  const [mode, setMode] = useState<"palette" | "css">("palette");
   547→  const [colors, setColors] = useState<ColorRow[]>(() =>
   548→    DEFAULT_COLORS.map((c) => ({ ...c })),
   549→  );
   550→  const [cssInput, setCssInput] = useState(DEFAULT_CSS);

   552→  const [parseError, setParseError] = useState<string | null>(null);
   553→
   554→  const [hueRotation, setHueRotation] = useState(false);
   555→  const [boostContrast, setBoostContrast] = useState(30);
   556→  const [swapColumns, setSwapColumns] = useState(false);
   557→  const [previewMode, setPreviewMode] = useState<"light" | "dark">("dark");

   561→  const conversions = useMemo<ConversionResult[]>(() => {
   562→    return colors.map((row): ConversionResult => {
   563→      const inputOklch = hexToOklch(row.hex);
   564→      if (!inputOklch) {
   565→        return { row, inputOklch: null, output: null, outputHex: "" };
   566→      }
   567→      const output = convertToDark(inputOklch, { hueRotation, boostContrast });
   568→      const outputHex = oklchToHex(output);
   569→      return { row, inputOklch, output, outputHex };
   570→    });
   571→  }, [colors, hueRotation, boostContrast]);
   572→
   573→  /* ─── Dark bg reference for contrast indicator ─── */

   575→    const valid = conversions.filter((c) => c.inputOklch && c.outputHex);
   576→    if (valid.length === 0) return null;
   577→    // Prefer a row labeled "background"/"bg"
   578→    const bgRow = valid.find((c) => /bg|background/i.test(c.row.label));
   579→    if (bgRow?.outputHex) return bgRow.outputHex;
   580→    // Fallback: the lightest input's dark conversion (becomes the dark bg)
   581→    const lightest = [...valid].sort(
   582→      (a, b) => (b.inputOklch?.L ?? 0) - (a.inputOklch?.L ?? 0),
   583→    )[0];
   584→    return lightest?.outputHex ?? null;
   585→  }, [conversions]);
   586→
   587→  const darkBgRgb = useMemo<[number, number, number] | null>(() => {
   588→    if (!darkBgHex) return null;

   590→  }, [darkBgHex]);

   592→  /* ─── Live preview palette (light = input, dark = generated) ─── */

   594→    const source: { hex: string; label: string }[] =
   595→      previewMode === "light"
   596→        ? colors.map((c) => ({ hex: c.hex, label: c.label }))
   597→        : conversions
   598→            .filter((c) => c.outputHex)
   599→            .map((c) => ({ hex: c.outputHex, label: c.row.label }));
   600→
   601→    const find = (...roles: string[]) => {
   602→      for (const role of roles) {
   603→        const hit = source.find((c) =>
   604→          c.label.toLowerCase().includes(role.toLowerCase()),
   605→        );
   606→        if (hit) return hit.hex;
   607→      }
   608→      return null;
   609→    };
   610→
   611→    return {
   612→      bg: find("background", "bg"),
   613→      surface: find("surface", "card"),
   614→      text: find("text", "foreground", "fg"),
   615→      muted: find("muted", "secondary"),
   616→      primary: find("primary", "accent", "brand"),
   617→      fallback: source[0]?.hex ?? "#ffffff",
   618→    };
   619→  }, [previewMode, colors, conversions]);
   620→
   621→  /* ─── Export payloads (dark palette) ─── */

   651→      },
   652→    ]);
   653→  }, []);

   655→  const removeRow = useCallback((id: string) => {

   657→  }, []);

   663→    setParseError(null);

   665→

   668→    setCssInput("");

   670→    setParseError(null);

   722→          <MoonStar className="size-4" />

   725→          <h3 className="font-display text-base font-semibold text-foreground leading-tight">
   726→            Dark Mode Color Converter
   727→          </h3>
   728→          <p className="text-xs text-muted-foreground">
   729→            Generate a perceptually-tuned dark palette from your light-mode colors via OKLCH.
   730→          </p>
   731→        </div>
   732→      </div>
   733→
   734→      {/* Controls bar */}

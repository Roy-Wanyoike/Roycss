    72→/* ─── Scroll Progress Bar ───────────────────────────────────── */
    73→function ScrollProgress() {
    74→  const { scrollYProgress } = useScroll();
    75→  const scaleX = useSpring(scrollYProgress, {
    76→    stiffness: 120,
    77→    damping: 30,
    78→    restDelta: 0.001,
    79→  });
    80→  return (
    81→    <motion.div
    82→      style={{ scaleX }}
    83→      className="roycss-scroll-progress"
    84→    />
    85→  );
    86→}

    88→/* ─── Theme Toggle ──────────────────────────────────────────── */
    89→function ThemeToggle() {
    90→  const [dark, setDark] = useState(true);
    91→
    92→  useEffect(() => {
    93→    const root = document.documentElement;
    94→    if (dark) {
    95→      root.classList.add("dark");
    96→    } else {
    97→      root.classList.remove("dark");
    98→    }
    99→  }, [dark]);
   100→
   101→  return (
   102→    <button
   103→      onClick={() => setDark(!dark)}
   104→      className="flex items-center justify-center size-9 rounded-xl glass text-muted-foreground hover:text-foreground transition-all hover:-translate-y-0.5 cursor-pointer"
   105→      aria-label="Toggle theme"
   106→    >
   107→      {dark ? <Sun className="size-4" /> : <Moon className="size-4" />}
   108→    </button>
   109→  );
   110→}

   148→/* ─── Install Command ───────────────────────────────────────── */
   149→function InstallCommand() {
   150→  const [copied, setCopied] = useState(false);
   151→  const handleCopy = async () => {
   152→    try {
   153→      await navigator.clipboard.writeText("npm install roycss");
   154→      setCopied(true);
   155→      setTimeout(() => setCopied(false), 2000);

   158→    }
   159→  };
   160→
   161→  return (
   162→    <MagneticButton
   163→      strength={0.25}
   164→      className="inline-block"
   165→    >
   166→      <div
   167→        className="inline-flex items-center gap-2 rounded-xl glass-strong px-4 py-2.5 group cursor-pointer hover:border-primary/30 transition-all"

   170→        <span className="text-sm font-mono text-muted-foreground">$</span>
   171→        <code className="text-sm font-mono text-foreground">npm install roycss</code>

   173→          className="text-muted-foreground hover:text-foreground transition-colors ml-2 cursor-pointer"

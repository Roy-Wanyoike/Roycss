    28→/**
    29→ * FluidTypographyCalculator — a CSS `clamp()` generator for fluid typography.
    30→ *
    31→ * Generates a `clamp(MIN, PREFERRED, MAX)` value that scales font-size
    32→ * linearly between a min and max viewport width, hitting exactly minFS at
    33→ * minVW and maxFS at maxVW.
    34→ *
    35→ * Math derivation:
    36→ *   We want `font = S + R * (1vw)` where `1vw` resolves to viewport/100 px.
    37→ *   At vw = minVW (so 1vw = minVW/100 px), font = minFS:
    38→ *     minFS = S + R * (minVW / 100)
    39→ *   At vw = maxVW, font = maxFS:
    40→ *     maxFS = S + R * (maxVW / 100)
    41→ *   Subtracting:  maxFS - minFS = R * (maxVW - minVW) / 100
    42→ *   =>  R = 100 * (maxFS - minFS) / (maxVW - minVW)   (vw coefficient, unitless)
    43→ *   =>  S = minFS - R * (minVW / 100)                  (intercept, in font units)
    44→ *
    45→ *   Equivalently, let B = (maxFS - minFS) / (maxVW - minVW)  (px per px):
    46→ *     R = 100 * B
    47→ *     S_px = minFS - B * minVW
    48→ *   For rem output, divide px values by rootFontSize (R is dimensionless).
    49→ *
    50→ * Live preview uses one iframe per simulated viewport — the iframe's layout
    51→ * width equals the viewport width, so `vw` resolves correctly inside it.
    52→ */
    53→

   109→// ─── Math helpers ───────────────────────────────────────────────────────────

   111→function clampNum(n: number, lo: number, hi: number): number {
   112→  return Math.min(hi, Math.max(lo, n));
   113→}
   114→
   115→/** Format a number with up to `decimals` digits, trimming trailing zeros. */
   116→function fmt(n: number, decimals = 2): string {
   117→  return Number(n.toFixed(decimals)).toString();
   118→}
   119→
   120→function computeClamp(
   121→  minFS: number,
   122→  maxFS: number,
   123→  minVW: number,
   124→  maxVW: number,
   125→  unit: Unit,
   126→  rootFontSize: number,
   127→): ClampParams {
   128→  // Guard against degenerate inputs (division by zero / reversed ranges).
   129→  const vRange = maxVW - minVW;
   130→  const safeRange = vRange === 0 ? 1 : vRange;
   131→  const B = (maxFS - minFS) / safeRange;
   132→  const R = 100 * B;
   133→  const Spx = minFS - B * minVW;
   134→
   135→  if (unit === "rem") {
   136→    const root = rootFontSize || 16;
   137→    const minStr = `${fmt(minFS / root)}rem`;
   138→    const maxStr = `${fmt(maxFS / root)}rem`;
   139→    const preferredStr = `${fmt(Spx / root)}rem + ${fmt(R, 3)}vw`;
   140→    return {
   141→      B,
   142→      R,
   143→      Spx,
   144→      S: Spx / root,
   145→      minStr,
   146→      maxStr,
   147→      preferredStr,
   148→      fullClamp: `clamp(${minStr}, ${preferredStr}, ${maxStr})`,
   149→    };
   150→  }
   151→
   152→  const minStr = `${fmt(minFS)}px`;
   153→  const maxStr = `${fmt(maxFS)}px`;
   154→  const preferredStr = `${fmt(Spx)}px + ${fmt(R, 3)}vw`;
   155→  return {
   156→    B,
   157→    R,
   158→    Spx,
   159→    S: Spx,
   160→    minStr,
   161→    maxStr,
   162→    preferredStr,
   163→    fullClamp: `clamp(${minStr}, ${preferredStr}, ${maxStr})`,
   164→  };
   165→}
   166→
   167→/** Resolved font size (px) at a given viewport width, clamping outside the ramp. */
   168→function fontSizeAtV(
   169→  V: number,
   170→  minFS: number,
   171→  maxFS: number,
   172→  minVW: number,
   173→  maxVW: number,
   174→): number {
   175→  if (V <= minVW) return minFS;
   176→  if (V >= maxVW) return maxFS;
   177→  const range = maxVW - minVW;
   178→  if (range === 0) return minFS;
   179→  return minFS + ((V - minVW) * (maxFS - minFS)) / range;
   180→}
   181→
   182→/** Escape HTML special characters in user-provided text. */
   183→function escapeHtml(s: string): string {
   184→  return s
   185→    .replace(/&/g, "&amp;")
   186→    .replace(/</g, "&lt;")
   187→    .replace(/>/g, "&gt;")
   188→    .replace(/"/g, "&quot;")
   189→    .replace(/'/g, "&#39;");
   190→}

   287→function CurveChart({
   288→  minFS,
   289→  maxFS,
   290→  minVW,
   291→  maxVW,
   292→  invalid,
   293→}: CurveChartProps) {
   294→  const W = 280;
   295→  const H = 124;
   296→  const padL = 34;
   297→  const padR = 12;
   298→  const padT = 10;
   299→  const padB = 24;
   300→  const plotW = W - padL - padR;
   301→  const plotH = H - padT - padB;
   302→
   303→  // Normalize so the chart stays sane even when inputs are reversed.
   304→  const loFS = Math.min(minFS, maxFS);
   305→  const hiFS = Math.max(minFS, maxFS);
   306→  const loVW = Math.min(minVW, maxVW);
   307→  const hiVW = Math.max(minVW, maxVW);
   308→
   309→  const xMax = Math.max(1600, Math.ceil((hiVW + 100) / 100) * 100);
   310→  const yMaxRaw = Math.max(hiFS * 1.15, loFS + 4);
   311→  const yMax = Math.max(8, Math.ceil(yMaxRaw / 4) * 4);
   312→
   313→  const xScale = (v: number) => padL + (v / xMax) * plotW;
   314→  const yScale = (f: number) => padT + plotH - (f / yMax) * plotH;
   315→
   316→  const pts = [
   317→    { v: 0, f: loFS },
   318→    { v: loVW, f: loFS },
   319→    { v: hiVW, f: hiFS },
   320→    { v: xMax, f: hiFS },
   321→  ];
   322→  const path = pts
   323→    .map(
   324→      (p, i) =>
   325→        `${i === 0 ? "M" : "L"} ${xScale(p.v).toFixed(2)} ${yScale(p.f).toFixed(2)}`,
   326→    )
   327→    .join(" ");
   328→
   329→  const midV = (loVW + hiVW) / 2;
   330→  const midF = (loFS + hiFS) / 2;
   331→
   332→  const xTicks = [0, loVW, hiVW, xMax];
   333→  const ariaLabel = invalid
   334→    ? "Font-size curve is invalid: max font size is below min, or max viewport is not greater than min."
   335→    : `Font-size curve: flat at ${fmt(loFS)}px until viewport ${loVW}px, ramps linearly to ${fmt(hiFS)}px at ${hiVW}px viewport, then flat.`;
   336→
   337→  return (
   338→    <div className="space-y-1 rounded-xl border border-border bg-card p-4">
   339→      <div className="flex items-center justify-between">
   340→        <span className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   341→          Font-size curve
   342→        </span>
   343→        <span className="font-mono text-[10px] text-muted-foreground">
   344→          px ↑ / px →
   345→        </span>
   346→      </div>
   347→      <svg
   348→        viewBox={`0 0 ${W} ${H}`}
   349→        className="w-full"
   350→        role="img"
   351→        aria-label={ariaLabel}
   352→      >
   353→        {/* Horizontal grid lines at loFS / hiFS */}
   354→        <line
   355→          x1={padL}
   356→          y1={yScale(loFS)}
   357→          x2={W - padR}
   358→          y2={yScale(loFS)}
   359→          className="stroke-muted-foreground/25"
   360→          strokeWidth={0.5}
   361→          strokeDasharray="2 3"
   362→        />
   363→        <line
   364→          x1={padL}
   365→          y1={yScale(hiFS)}
   366→          x2={W - padR}
   367→          y2={yScale(hiFS)}
   368→          className="stroke-muted-foreground/25"
   369→          strokeWidth={0.5}
   370→          strokeDasharray="2 3"
   371→        />
   372→
   373→        {/* Vertical guide lines at loVW / hiVW */}
   374→        <line
   375→          x1={xScale(loVW)}
   376→          y1={padT}
   377→          x2={xScale(loVW)}
   378→          y2={padT + plotH}
   379→          className="stroke-muted-foreground/25"
   380→          strokeWidth={0.5}
   381→          strokeDasharray="2 3"
   382→        />
   383→        <line
   384→          x1={xScale(hiVW)}
   385→          y1={padT}
   386→          x2={xScale(hiVW)}
   387→          y2={padT + plotH}
   388→          className="stroke-muted-foreground/25"
   389→          strokeWidth={0.5}
   390→          strokeDasharray="2 3"
   391→        />
   392→
   393→        {/* Axes */}
   394→        <line
   395→          x1={padL}
   396→          y1={padT}
   397→          x2={padL}
   398→          y2={padT + plotH}
   399→          className="stroke-border"
   400→          strokeWidth={1}
   401→        />
   402→        <line
   403→          x1={padL}
   404→          y1={padT + plotH}
   405→          x2={W - padR}
   406→          y2={padT + plotH}
   407→          className="stroke-border"
   408→          strokeWidth={1}
   409→        />
   410→
   411→        {/* The ramp path (flat → linear → flat) */}
   412→        <path
   413→          d={path}
   414→          fill="none"
   415→          className="stroke-primary"
   416→          strokeWidth={2}
   417→          strokeLinejoin="round"
   418→          strokeLinecap="round"
   419→        />
   420→
   421→        {/* Midpoint of the ramp (open dot) */}
   422→        <circle
   423→          cx={xScale(midV)}
   424→          cy={yScale(midF)}
   425→          r={2}
   426→          className="fill-background stroke-primary"
   427→          strokeWidth={1.5}
   428→        />
   429→
   430→        {/* Min/Max knee points (filled dots) */}
   431→        <circle
   432→          cx={xScale(loVW)}
   433→          cy={yScale(loFS)}
   434→          r={3.5}
   435→          className="fill-primary"
   436→        >
   437→          <title>{`min: ${fmt(loFS)}px @ ${loVW}px`}</title>
   438→        </circle>
   439→        <circle
   440→          cx={xScale(hiVW)}
   441→          cy={yScale(hiFS)}
   442→          r={3.5}
   443→          className="fill-primary"
   444→        >
   445→          <title>{`max: ${fmt(hiFS)}px @ ${hiVW}px`}</title>
   446→        </circle>
   447→
   448→        {/* X-axis tick labels */}
   449→        {xTicks.map((v) => (
   450→          <text
   451→            key={`x-${v}`}
   452→            x={xScale(v)}
   453→            y={padT + plotH + 12}
   454→            textAnchor="middle"
   455→            className="fill-muted-foreground"
   456→            fontSize={8}
   457→            fontFamily="ui-monospace, SFMono-Regular, Menlo, monospace"
   458→          >
   459→            {v}
   460→          </text>
   461→        ))}
   462→
   463→        {/* Y-axis tick labels */}
   464→        <text
   465→          x={padL - 4}
   466→          y={yScale(loFS) + 3}
   467→          textAnchor="end"
   468→          className="fill-muted-foreground"
   469→          fontSize={8}
   470→          fontFamily="ui-monospace, SFMono-Regular, Menlo, monospace"
   471→        >
   472→          {fmt(loFS)}
   473→        </text>
   474→        <text
   475→          x={padL - 4}
   476→          y={yScale(hiFS) + 3}
   477→          textAnchor="end"
   478→          className="fill-muted-foreground"
   479→          fontSize={8}
   480→          fontFamily="ui-monospace, SFMono-Regular, Menlo, monospace"
   481→        >
   482→          {fmt(hiFS)}
   483→        </text>
   484→
   485→        {/* Axis title */}
   486→        <text
   487→          x={(padL + W - padR) / 2}
   488→          y={H - 2}
   489→          textAnchor="middle"
   490→          className="fill-muted-foreground"
   491→          fontSize={8}
   492→        >
   493→          viewport width →
   494→        </text>
   495→      </svg>
   496→    </div>
   497→  );
   498→}

   502→export function FluidTypographyCalculator() {
   503→  // ── State: inputs ─────────────────────────────────────────────────────

   505→  const [maxFS, setMaxFS] = useState(32);

   507→  const [maxVW, setMaxVW] = useState(1200);
   508→  const [unit, setUnit] = useState<Unit>("rem");
   509→  const [rootFontSize, setRootFontSize] = useState(16);
   510→  const [fontFamily, setFontFamily] = useState("system-ui, sans-serif");
   511→  const [previewText, setPreviewText] = useState(
   512→    "The quick brown fox jumps over the lazy dog",
   513→  );
   514→
   515→  // ── State: UI ─────────────────────────────────────────────────────────
   516→  const [copied, setCopied] = useState<CopyTarget | null>(null);
   517→  const [themeColors, setThemeColors] = useState<ThemeColors | null>(null);
   518→
   519→  // Debounced inputs that feed the iframe srcdoc rebuild.
   520→  const [debounced, setDebounced] = useState({
   521→    text: previewText,
   522→    family: fontFamily,
   523→    fullClamp: "clamp(1rem, 0.64rem + 1.82vw, 2rem)",
   524→  });
   525→
   526→  const copyTimerRef = useRef<number | null>(null);
   527→
   528→  // ── Derived: clamp parameters & CSS strings ───────────────────────────
   529→  const params = useMemo(
   530→    () => computeClamp(minFS, maxFS, minVW, maxVW, unit, rootFontSize),
   531→    [minFS, maxFS, minVW, maxVW, unit, rootFontSize],
   532→  );
   533→
   534→  const cssRule = `font-size: ${params.fullClamp};`;
   535→  const cssFull = `font-family: ${fontFamily};\nfont-size: ${params.fullClamp};`;
   536→
   537→  const maxFSLowerThanMin = maxFS < minFS;
   538→  const maxVWNotGreater = maxVW <= minVW;
   539→  const configInvalid = maxFSLowerThanMin || maxVWNotGreater;
   540→
   541→  // ── Effect: track theme color changes (for iframe styling) ────────────
   542→  // iframes are isolated documents — they don't inherit CSS variables from
   543→  // the parent. We read the resolved values of --background / --foreground
   544→  // / --muted-foreground / --border from the root element and inject them
   545→  // explicitly into each iframe's srcdoc. A MutationObserver keeps them in
   546→  // sync when the user toggles between light/dark theme.
   547→  useEffect(() => {
   548→    if (typeof window === "undefined") return;
   549→    const root = document.documentElement;
   550→    const read = () => {
   551→      const cs = getComputedStyle(root);
   552→      const bg = cs.getPropertyValue("--background").trim();
   553→      const fg = cs.getPropertyValue("--foreground").trim();
   554→      const muted = cs.getPropertyValue("--muted-foreground").trim();
   555→      const border = cs.getPropertyValue("--border").trim();
   556→      if (bg && fg) {
   557→        setThemeColors({
   558→          bg,
   559→          fg,
   560→          muted: muted || fg,
   561→          border: border || muted || fg,
   562→        });
   563→      }
   564→    };
   565→    read();
   566→    const obs = new MutationObserver(read);
   567→    obs.observe(root, { attributes: true, attributeFilter: ["class"] });
   568→    return () => obs.disconnect();
   569→  }, []);
   570→
   571→  // ── Effect: debounce iframe srcdoc rebuild (~200ms) ───────────────────
   572→  // Rebuilding 6 iframes on every keystroke is wasteful and flickery; we
   573→  // debounce so rapid slider drags / typing only trigger one rebuild.
   574→  useEffect(() => {
   575→    const t = window.setTimeout(() => {
   576→      setDebounced({
   577→        text: previewText,
   578→        family: fontFamily,
   579→        fullClamp: params.fullClamp,
   580→      });
   581→    }, IFRAME_DEBOUNCE_MS);
   582→    return () => window.clearTimeout(t);
   583→  }, [previewText, fontFamily, params.fullClamp]);
   584→
   585→  // ── Effect: cleanup copy-confirmation timer on unmount ────────────────

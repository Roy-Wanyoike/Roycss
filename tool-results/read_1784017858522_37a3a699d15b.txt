   102→/* ─── Scroll Progress Bar ───────────────────────────────────── */
   103→function ScrollProgress() {
   104→  const { scrollYProgress } = useScroll();
   105→  const scaleX = useSpring(scrollYProgress, {
   106→    stiffness: 120,
   107→    damping: 30,
   108→    restDelta: 0.001,
   109→  });
   110→  return (
   111→    <motion.div
   112→      style={{ scaleX }}
   113→      className="roycss-scroll-progress"
   114→    />
   115→  );
   116→}

   118→/* ─── Theme Toggle ──────────────────────────────────────────── */
   119→function ThemeToggle() {
   120→  const [dark, setDark] = useState(true);
   121→
   122→  useEffect(() => {
   123→    const root = document.documentElement;
   124→    if (dark) {
   125→      root.classList.add("dark");
   126→    } else {
   127→      root.classList.remove("dark");
   128→    }
   129→  }, [dark]);
   130→
   131→  return (
   132→    <button
   133→      onClick={() => setDark(!dark)}
   134→      className="flex items-center justify-center size-9 rounded-xl glass text-muted-foreground hover:text-foreground transition-all hover:-translate-y-0.5 cursor-pointer"
   135→      aria-label="Toggle theme"
   136→    >
   137→      {dark ? <Sun className="size-4" /> : <Moon className="size-4" />}
   138→    </button>
   139→  );
   140→}

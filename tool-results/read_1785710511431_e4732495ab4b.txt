    30→/**
    31→ * GridAreasBuilder — a visual editor for the CSS `grid-template-areas` property.
    32→ *
    33→ * The user draws a 2D "map" of named areas by painting cells with a current
    34→ * brush (selected from the palette) or by double-clicking a cell to rename it
    35→ * inline. Drag-select paints a rectangular range with the current brush.
    36→ *
    37→ * Output:
    38→ *   - A live, read-only CSS block (`display: grid; grid-template-columns: …;
    39→ *     grid-template-rows: …; grid-template-areas: "…" "…" …;`).
    40→ *   - A live preview that actually renders the grid using the generated CSS.
    41→ *
    42→ * Validation surfaces two classes of problems as amber badges:
    43→ *   - Invalid identifiers (names not matching `^[A-Za-z][A-Za-z0-9_-]*$`).
    44→ *   - Non-rectangular regions (a single name whose cells do not form a single
    45→ *     solid rectangle — CSS requires this).
    46→ *
    47→ * Interaction model (resolves the spec's click-to-paint vs click-to-edit
    48→ * tension):
    49→ *   - Single click on a cell  → paint with the current brush.
    50→ *   - Double-click on a cell  → edit its name inline (Enter/blur commits,
    51→ *                                Escape cancels, empty/`.` clears).
    52→ *   - Mousedown + drag + up   → paint the rectangular range with the brush.
    53→ *
    54→ * The component is fully self-contained and uses only semantic Tailwind theme
    55→ * tokens for chrome (no indigo/blue, no hardcoded brand colors). The data-viz
    56→ * area tints use the eight-color palette at `/15` opacity, per spec.
    57→ */

   225→/** Valid CSS identifier for a grid-area name: letter followed by letters/digits/hyphens/underscores. */
   226→const IDENT_RE = /^[A-Za-z][A-Za-z0-9_-]*$/;
   227→
   228→function isValidAreaName(name: string): boolean {
   229→  return IDENT_RE.test(name);
   230→}

   232→/** Build a rows×cols grid, preserving old cell values where they still fit. */
   233→function resizeGrid(prev: Grid, rows: number, cols: number): Grid {
   234→  const next: Grid = [];
   235→  for (let r = 0; r < rows; r++) {
   236→    const row: string[] = [];
   237→    for (let c = 0; c < cols; c++) {
   238→      row.push(prev[r]?.[c] ?? DOT);
   239→    }
   240→    next.push(row);
   241→  }
   242→  return next;
   243→}
   244→
   245→/** Create a fresh rows×cols grid filled with the dot sentinel. */
   246→function emptyGrid(rows: number, cols: number): Grid {
   247→  return resizeGrid([], rows, cols);
   248→}

   250→/** Order-preserving list of unique non-dot names currently in the grid. */
   251→function getUniqueNames(grid: Grid): string[] {
   252→  const seen = new Set<string>();
   253→  const out: string[] = [];
   254→  for (const row of grid) {
   255→    for (const cell of row) {
   256→      if (cell !== DOT && !seen.has(cell)) {
   257→        seen.add(cell);
   258→        out.push(cell);
   259→      }
   260→    }
   261→  }
   262→  return out;
   263→}
   264→
   265→/** Stable tint lookup — index by the name's position in the unique-names list. */
   266→function tintForName(name: string, names: string[]): Tint {
   267→  const idx = names.indexOf(name);
   268→  return AREA_TINTS[(idx < 0 ? 0 : idx) % AREA_TINTS.length];
   269→}
   270→
   271→/** All coordinates currently occupied by `name`. */
   272→function cellsForName(grid: Grid, name: string): Cell[] {
   273→  const out: Cell[] = [];
   274→  for (let r = 0; r < grid.length; r++) {
   275→    const row = grid[r];
   276→    for (let c = 0; c < row.length; c++) {
   277→      if (row[c] === name) out.push({ r, c });
   278→    }
   279→  }
   280→  return out;
   281→}

   283→/**
   284→ * A named area is CSS-valid only if its cells form a single solid rectangle:
   285→ * the bounding box's area must equal the cell count AND every cell inside the
   286→ * bounding box must belong to `name`.
   287→ */
   288→function isRectangular(grid: Grid, name: string): boolean {
   289→  const cells = cellsForName(grid, name);
   290→  if (cells.length === 0) return true;
   291→  let minR = Infinity,
   292→    maxR = -Infinity,
   293→    minC = Infinity,
   294→    maxC = -Infinity;
   295→  for (const { r, c } of cells) {
   296→    if (r < minR) minR = r;
   297→    if (r > maxR) maxR = r;
   298→    if (c < minC) minC = c;
   299→    if (c > maxC) maxC = c;
   300→  }
   301→  const expected = (maxR - minR + 1) * (maxC - minC + 1);
   302→  if (cells.length !== expected) return false;
   303→  for (let r = minR; r <= maxR; r++) {
   304→    for (let c = minC; c <= maxC; c++) {
   305→      if (grid[r]?.[c] !== name) return false;
   306→    }
   307→  }
   308→  return true;
   309→}

   311→/** Generate the indented `"a b c"` block for `grid-template-areas:`. */
   312→function areasBlock(grid: Grid): string {
   313→  return grid.map((row) => `    "${row.join(" ")}"`).join("\n");
   314→}
   315→
   316→/** Generate the full CSS snippet. */
   317→function generateCss(
   318→  grid: Grid,
   319→  cols: number,
   320→  rows: number,
   321→  colValue: string,
   322→  rowValue: string
   323→): string {
   324→  return [
   325→    ".layout {",
   326→    "  display: grid;",
   327→    `  grid-template-columns: ${colValue};`,
   328→    `  grid-template-rows: ${rowValue};`,
   329→    "  grid-template-areas:",
   330→    areasBlock(grid) + ";",
   331→    "}",
   332→  ].join("\n");
   333→}

   335→/** Compute the column CSS value from the selected sizing + dims. */
   336→function resolveColValue(
   337→  sizing: string,
   338→  cols: number,
   339→  custom: string
   340→): string {
   341→  if (sizing === "custom") return custom.trim() || `repeat(${cols}, 1fr)`;
   342→  if (sizing === "auto") return `repeat(${cols}, auto)`;
   343→  return `repeat(${cols}, ${sizing})`;
   344→}
   345→
   346→/** Compute the row CSS value from the selected sizing + dims. */
   347→function resolveRowValue(
   348→  sizing: string,
   349→  rows: number,
   350→  custom: string
   351→): string {
   352→  if (sizing === "custom") return custom.trim() || `repeat(${rows}, auto)`;
   353→  if (sizing === "auto") return `repeat(${rows}, auto)`;
   354→  return `repeat(${rows}, ${sizing})`;
   355→}

   357→/** Build the `gridTemplateAreas` CSS string for the live preview. */
   358→function previewTemplateAreas(grid: Grid): string {
   359→  return grid.map((row) => `"${row.join(" ")}"`).join(" ");
   360→}

   362→/** Title-case a name for the live preview label ("sidebar" → "Sidebar"). */
   363→function prettyLabel(name: string): string {
   364→  if (!name) return "";
   365→  return name.charAt(0).toUpperCase() + name.slice(1);
   366→}
   367→
   368→/** Auto-name a new area: area-1, area-2, … skipping names already in use. */
   369→function nextAutoName(taken: string[]): string {
   370→  let n = 1;
   371→  while (taken.includes(`area-${n}`)) n++;
   372→  return `area-${n}`;
   373→}

   424→export function GridAreasBuilder() {
   425→  // Grid state ---------------------------------------------------------------
   426→  const [rows, setRows] = useState(3);
   427→  const [cols, setCols] = useState(3);
   428→  const [grid, setGrid] = useState<Grid>(() => PRESETS[0].grid.map((r) => [...r]));
   429→
   430→  // Brush + palette ----------------------------------------------------------
   431→  const [brush, setBrush] = useState<string>(DOT);
   432→
   433→  // Sizing -------------------------------------------------------------------
   434→  const [colSizing, setColSizing] = useState<string>("1fr");
   435→  const [rowSizing, setRowSizing] = useState<string>("auto");
   436→  const [customCol, setCustomCol] = useState<string>("200px 1fr 200px");
   437→  const [customRow, setCustomRow] = useState<string>("auto 1fr auto");
   438→
   439→  // Inline editing -----------------------------------------------------------
   440→  const [editing, setEditing] = useState<Cell | null>(null);
   441→  const [editValue, setEditValue] = useState<string>("");
   442→  const editInputRef = useRef<HTMLInputElement | null>(null);
   443→
   444→  // Drag-select --------------------------------------------------------------
   445→  const [isDragging, setIsDragging] = useState(false);
   446→  const [dragStart, setDragStart] = useState<Cell | null>(null);
   447→  const [dragEnd, setDragEnd] = useState<Cell | null>(null);
   448→
   449→  // Copy feedback ------------------------------------------------------------
   450→  const [copied, setCopied] = useState(false);
   451→
   452→  // Derived ------------------------------------------------------------------
   453→  const uniqueNames = useMemo(() => getUniqueNames(grid), [grid]);
   454→  const invalidNames = useMemo(
   455→    () => uniqueNames.filter((n) => !isValidAreaName(n)),
   456→    [uniqueNames]
   457→  );
   458→  const nonRectangular = useMemo(
   459→    () => uniqueNames.filter((n) => !isRectangular(grid, n)),
   460→    [grid, uniqueNames]
   461→  );
   462→  const colValue = useMemo(
   463→    () => resolveColValue(colSizing, cols, customCol),
   464→    [colSizing, cols, customCol]
   465→  );
   466→  const rowValue = useMemo(
   467→    () => resolveRowValue(rowSizing, rows, customRow),
   468→    [rowSizing, rows, customRow]
   469→  );
   470→  const css = useMemo(
   471→    () => generateCss(grid, cols, rows, colValue, rowValue),
   472→    [grid, cols, rows, colValue, rowValue]
   473→  );
   474→
   475→  // Focus the edit input whenever it appears --------------------------------
   476→  useEffect(() => {
   477→    if (editing && editInputRef.current) {
   478→      editInputRef.current.focus();
   479→      editInputRef.current.select();
   480→    }
   481→  }, [editing]);
   482→
   483→  // Global mouseup to commit drag-select ------------------------------------
   484→  useEffect(() => {
   485→    if (!isDragging) return;
   486→    const handleUp = () => {
   487→      if (dragStart && dragEnd) {
   488→        const minR = Math.min(dragStart.r, dragEnd.r);
   489→        const maxR = Math.max(dragStart.r, dragEnd.r);
   490→        const minC = Math.min(dragStart.c, dragEnd.c);
   491→        const maxC = Math.max(dragStart.c, dragEnd.c);
   492→        setGrid((prev) =>
   493→          prev.map((row, r) =>
   494→            r >= minR && r <= maxR
   495→              ? row.map((cell, c) =>
   496→                  c >= minC && c <= maxC ? brush : cell
   497→                )
   498→              : row
   499→          )
   500→        );
   501→      }
   502→      setIsDragging(false);
   503→      setDragStart(null);
   504→      setDragEnd(null);
   505→    };
   506→    window.addEventListener("mouseup", handleUp);
   507→    return () => window.removeEventListener("mouseup", handleUp);
   508→  }, [isDragging, dragStart, dragEnd, brush]);
   509→
   510→  // Mutators -----------------------------------------------------------------
   511→  const setDims = useCallback(
   512→    (nextRows: number, nextCols: number) => {
   513→      setRows(nextRows);
   514→      setCols(nextCols);
   515→      setGrid((prev) => resizeGrid(prev, nextRows, nextCols));
   516→    },
   517→    []
   518→  );
   519→
   520→  const startEdit = useCallback((cell: Cell, current: string) => {
   521→    setEditing(cell);
   522→    setEditValue(current === DOT ? "" : current);
   523→  }, []);
   524→
   525→  const commitEdit = useCallback(() => {
   526→    if (!editing) return;
   527→    const raw = editValue.trim();
   528→    const final = raw === "" ? DOT : raw;
   529→    setGrid((prev) =>
   530→      prev.map((row, r) =>
   531→        r === editing.r
   532→          ? row.map((cell, c) => (c === editing.c ? final : cell))
   533→          : row
   534→      )
   535→    );
   536→    setEditing(null);
   537→    setEditValue("");
   538→  }, [editing, editValue]);
   539→
   540→  const cancelEdit = useCallback(() => {
   541→    setEditing(null);
   542→    setEditValue("");
   543→  }, []);
   544→
   545→  const loadPreset = useCallback(
   546→    (preset: Preset) => {
   547→      const next = preset.grid.map((r) => [...r]);
   548→      setGrid(next);
   549→      setRows(next.length);
   550→      setCols(next[0]?.length ?? 1);
   551→      setBrush(DOT);
   552→      setEditing(null);
   553→    },
   554→    []
   555→  );
   556→
   557→  const clearGrid = useCallback(() => {
   558→    setGrid(emptyGrid(rows, cols));
   559→    setBrush(DOT);
   560→    setEditing(null);
   561→  }, [rows, cols]);
   562→
   563→  const addNewArea = useCallback(() => {
   564→    const name = nextAutoName(uniqueNames);
   565→    setBrush(name);
   566→    // Don't paint anything yet — the user picks where to put it.
   567→  }, [uniqueNames]);
   568→
   569→  const deleteArea = useCallback((name: string) => {
   570→    setGrid((prev) =>
   571→      prev.map((row) => row.map((cell) => (cell === name ? DOT : cell)))
   572→    );
   573→    setBrush((b) => (b === name ? DOT : b));
   574→  }, []);
   575→
   576→  const handleCopy = useCallback(async () => {
   577→    try {
   578→      await navigator.clipboard.writeText(css);
   579→      setCopied(true);
   580→      window.setTimeout(() => setCopied(false), 1600);
   581→    } catch {
   582→      // Clipboard may be unavailable (permissions, SSR). Silently no-op.
   583→      setCopied(false);
   584→    }
   585→  }, [css]);
   586→
   587→  // Drag range predicate -----------------------------------------------------
   588→  const inDragRange = useCallback(
   589→    (r: number, c: number): boolean => {
   590→      if (!isDragging || !dragStart || !dragEnd) return false;
   591→      const minR = Math.min(dragStart.r, dragEnd.r);

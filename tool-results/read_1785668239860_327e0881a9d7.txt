    17→    17→/**
    18→    18→ * EasingVisualizer — an interactive cubic-bezier easing curve designer.
    19→    19→ *
    20→    20→ * Features:
    21→    21→ *  - SVG curve editor with two draggable control points (pointer + keyboard).
    22→    22→ *  - Live moving dot driven by a `requestAnimationFrame` loop.
    23→    23→ *  - Bidirectionally synced numeric inputs + sliders (x1, y1, x2, y2).
    24→    24→ *  - Preset gallery (linear / ease / quad / cubic / back / bounce-ish) with
    25→    25→ *    mini curve thumbnails; active preset is highlighted.
    26→    26→ *  - Real-world live preview: a 48px circle traversing a track using the
    27→    27→ *    actual `cubic-bezier(...)` CSS transition.
    28→    28→ *  - Generated `transition-timing-function` CSS with copy-to-clipboard.
    29→    29→ *  - Configurable preview duration (100–5000ms).
    30→    30→ *
    31→    31→ * Math:
    32→    32→ *  - The cubic-bezier parametric form with P0=(0,0), P1=(x1,y1),
    33→    33→ *    P2=(x2,y2), P3=(1,1):
    34→    34→ *      B(t) = 3(1-t)^2 t P1 + 3(1-t) t^2 P2 + t^3 P3   (P0 term is 0)
    35→    35→ *  - For the moving dot we need y given x (time). Because x1,x2 ∈ [0,1],
    36→    36→ *    Bx(t) is monotonically increasing in t, so a 40-iteration binary
    37→    37→ *    search on t converges to sub-pixel precision.
    38→    38→ */
    39→    39→
    40→    40→interface BezierVals {
    41→    41→  x1: number;
    42→    42→  y1: number;
    43→    43→  x2: number;
    44→    44→  y2: number;
    45→    45→}

    49→    49→  /** Flag for approximate / non-standard easings (e.g. "bounce-ish"). */

    51→    51→}

    73→    73→/** SVG viewBox dimension for the graph square (in SVG user units). */
    74→    74→const VB = 280;
    75→    75→/** Y-axis clamp range (allows overshoot / back / bounce-style easings). */
    76→    76→const Y_MIN = -0.5;
    77→    77→const Y_MAX = 1.5;
    78→    78→
    79→    79→function clamp(n: number, lo: number, hi: number): number {
    80→    80→  return Math.min(hi, Math.max(lo, n));
    81→    81→}
    82→    82→
    83→    83→/** Trim trailing zeros so 0.5 stays "0.5" and 0.42 stays "0.42". */
    84→    84→function fmt(n: number): string {
    85→    85→  return Number(n.toFixed(3)).toString();
    86→    86→}
    87→    87→
    88→    88→/**
    89→    89→ * Solve `y = By(t)` for a given `x = Bx(t)` using binary search.
    90→    90→ * Requires x1, x2 ∈ [0,1] (guaranteed by clamping) so Bx is monotonic.
    91→    91→ */
    92→    92→function bezierYForX(
    93→    93→  x: number,
    94→    94→  x1: number,
    95→    95→  y1: number,
    96→    96→  x2: number,
    97→    97→  y2: number,
    98→    98→): number {
    99→    99→  let lo = 0;
   100→   100→  let hi = 1;
   101→   101→  for (let i = 0; i < 40; i++) {
   102→   102→    const t = (lo + hi) / 2;
   103→   103→    const mt = 1 - t;
   104→   104→    const bx = 3 * mt * mt * t * x1 + 3 * mt * t * t * x2 + t * t * t;
   105→   105→    if (bx < x) lo = t;
   106→   106→    else hi = t;
   107→   107→  }
   108→   108→  const t = (lo + hi) / 2;
   109→   109→  const mt = 1 - t;
   110→   110→  return 3 * mt * mt * t * y1 + 3 * mt * t * t * y2 + t * t * t;
   111→   111→}
   112→   112→
   113→   113→/** Sample the bezier into a polyline SVG path (robust, no curve command needed). */
   114→   114→function sampleCurvePath(
   115→   115→  x1: number,
   116→   116→  y1: number,
   117→   117→  x2: number,
   118→   118→  y2: number,
   119→   119→  samples: number,
   120→   120→  scale: number,
   121→   121→): string {
   122→   122→  const pts: string[] = [];
   123→   123→  for (let i = 0; i <= samples; i++) {
   124→   124→    const t = i / samples;
   125→   125→    const mt = 1 - t;
   126→   126→    const x = 3 * mt * mt * t * x1 + 3 * mt * t * t * x2 + t * t * t;
   127→   127→    const y = 3 * mt * mt * t * y1 + 3 * mt * t * t * y2 + t * t * t;
   128→   128→    pts.push(`${(x * scale).toFixed(2)},${((1 - y) * scale).toFixed(2)}`);
   129→   129→  }
   130→   130→  return `M ${pts.join(" L ")}`;
   131→   131→}

   225→   225→export function EasingVisualizer() {
   226→   226→  const [vals, setVals] = useState<BezierVals>(DEFAULT_VALS);
   227→   227→  const [duration, setDuration] = useState(1500);
   228→   228→  const [playing, setPlaying] = useState(true);
   229→   229→  const [progress, setProgress] = useState(0);
   230→   230→  const [previewPos, setPreviewPos] = useState<0 | 1>(0);
   231→   231→  const [copied, setCopied] = useState(false);
   232→   232→  const [dragging, setDragging] = useState<DragTarget | null>(null);
   233→   233→
   234→   234→  const svgRef = useRef<SVGSVGElement | null>(null);
   235→   235→  const trackRef = useRef<HTMLDivElement | null>(null);
   236→   236→  const [trackWidth, setTrackWidth] = useState(0);
   237→   237→
   238→   238→  const rafRef = useRef<number | null>(null);
   239→   239→  const lastTimeRef = useRef(0);
   240→   240→
   241→   241→  const { x1, y1, x2, y2 } = vals;
   242→   242→
   243→   243→  /* ── Track width (for live preview translate distance) ─────────────── */
   244→   244→  useEffect(() => {
   245→   245→    const el = trackRef.current;
   246→   246→    if (!el) return;
   247→   247→    const update = () => setTrackWidth(el.offsetWidth);
   248→   248→    update();
   249→   249→    const ro = new ResizeObserver(update);
   250→   250→    ro.observe(el);
   251→   251→    return () => ro.disconnect();
   252→   252→  }, []);
   253→   253→
   254→   254→  /* ── Moving dot: requestAnimationFrame loop, cleaned up on unmount ── */
   255→   255→  useEffect(() => {
   256→   256→    if (!playing) {
   257→   257→      lastTimeRef.current = 0;
   258→   258→      return;
   259→   259→    }
   260→   260→    const tick = (now: number) => {
   261→   261→      if (lastTimeRef.current === 0) lastTimeRef.current = now;
   262→   262→      const dt = now - lastTimeRef.current;
   263→   263→      lastTimeRef.current = now;
   264→   264→      setProgress((p) => {
   265→   265→        const next = p + dt / duration;
   266→   266→        // Loop seamlessly, preserving sub-frame precision.
   267→   267→        return next >= 1 ? next - Math.floor(next) : next;
   268→   268→      });
   269→   269→      rafRef.current = requestAnimationFrame(tick);
   270→   270→    };
   271→   271→    rafRef.current = requestAnimationFrame(tick);
   272→   272→    return () => {
   273→   273→      if (rafRef.current !== null) cancelAnimationFrame(rafRef.current);
   274→   274→      lastTimeRef.current = 0;
   275→   275→    };
   276→   276→  }, [playing, duration]);
   277→   277→
   278→   278→  /* ── Live preview box: CSS transition toggled on an interval ──────── */
   279→   279→  useEffect(() => {
   280→   280→    if (!playing) return;
   281→   281→    // Kick off the back-and-forth cycle on the next animation frame so we
   282→   282→    // don't trigger a synchronous re-render inside this effect body.
   283→   283→    const kickoff = requestAnimationFrame(() => setPreviewPos(1));
   284→   284→    const interval = window.setInterval(
   285→   285→      () => setPreviewPos((p) => (p === 1 ? 0 : 1)),
   286→   286→      duration + 120,
   287→   287→    );
   288→   288→    return () => {
   289→   289→      cancelAnimationFrame(kickoff);
   290→   290→      window.clearInterval(interval);
   291→   291→    };
   292→   292→  }, [playing, duration]);
   293→   293→
   294→   294→  const curvePath = useMemo(
   295→   295→    () => sampleCurvePath(x1, y1, x2, y2, 60, VB),
   296→   296→    [x1, y1, x2, y2],
   297→   297→  );
   298→   298→
   299→   299→  const easedY = useMemo(
   300→   300→    () => bezierYForX(progress, x1, y1, x2, y2),
   301→   301→    [progress, x1, y1, x2, y2],
   302→   302→  );
   303→   303→
   304→   304→  const cssString = `transition-timing-function: cubic-bezier(${fmt(x1)}, ${fmt(y1)}, ${fmt(x2)}, ${fmt(y2)});`;
   305→   305→  const cubicBezierString = `cubic-bezier(${fmt(x1)}, ${fmt(y1)}, ${fmt(x2)}, ${fmt(y2)})`;
   306→   306→
   307→   307→  const handleCopy = useCallback(async () => {
   308→   308→    try {
   309→   309→      await navigator.clipboard.writeText(cssString);
   310→   310→      setCopied(true);
   311→   311→      window.setTimeout(() => setCopied(false), 2000);
   312→   312→    } catch {
   313→   313→      /* clipboard unavailable — silently ignore */
   314→   314→    }
   315→   315→  }, [cssString]);
   316→   316→
   317→   317→  const setVal = useCallback(
   318→   318→    (key: keyof BezierVals, v: number) => {
   319→   319→      setVals((prev) => ({ ...prev, [key]: v }));
   320→   320→    },
   321→   321→    [],
   322→   322→  );
   323→   323→
   324→   324→  const handleReset = useCallback(() => {
   325→   325→    setVals(DEFAULT_VALS);
   326→   326→    setProgress(0);
   327→   327→  }, []);
   328→   328→
   329→   329→  /* ── Pointer → normalized (x, y) conversion using the SVG CTM ─────── */
   330→   330→  const getNormalized = useCallback(
   331→   331→    (e: React.PointerEvent): { x: number; y: number } | null => {
   332→   332→      const svg = svgRef.current;
   333→   333→      if (!svg) return null;
   334→   334→      const ctm = svg.getScreenCTM();
   335→   335→      if (!ctm) return null;
   336→   336→      const pt = new DOMPoint(e.clientX, e.clientY);
   337→   337→      const svgP = pt.matrixTransform(ctm.inverse());
   338→   338→      return {
   339→   339→        x: clamp(svgP.x / VB, 0, 1),
   340→   340→        y: clamp(1 - svgP.y / VB, Y_MIN, Y_MAX),
   341→   341→      };
   342→   342→    },
   343→   343→    [],
   344→   344→  );
   345→   345→
   346→   346→  const onPointerDownControl = useCallback(
   347→   347→    (e: React.PointerEvent<SVGGElement>, target: DragTarget) => {
   348→   348→      e.preventDefault();
   349→   349→      e.stopPropagation();
   350→   350→      (e.currentTarget as Element).setPointerCapture(e.pointerId);
   351→   351→      setDragging(target);
   352→   352→      const n = getNormalized(e);
   353→   353→      if (!n) return;
   354→   354→      if (target === "p1") {
   355→   355→        setVals((prev) => ({ ...prev, x1: n.x, y1: n.y }));
   356→   356→      } else {
   357→   357→        setVals((prev) => ({ ...prev, x2: n.x, y2: n.y }));
   358→   358→      }
   359→   359→    },
   360→   360→    [getNormalized],
   361→   361→  );
   362→   362→
   363→   363→  const onPointerMoveControl = useCallback(
   364→   364→    (e: React.PointerEvent<SVGGElement>) => {
   365→   365→      if (!dragging) return;
   366→   366→      const n = getNormalized(e);
   367→   367→      if (!n) return;
   368→   368→      if (dragging === "p1") {
   369→   369→        setVals((prev) => ({ ...prev, x1: n.x, y1: n.y }));
   370→   370→      } else {
   371→   371→        setVals((prev) => ({ ...prev, x2: n.x, y2: n.y }));
   372→   372→      }
   373→   373→    },
   374→   374→    [dragging, getNormalized],
   375→   375→  );
   376→   376→
   377→   377→  const onPointerUpControl = useCallback(
   378→   378→    (e: React.PointerEvent<SVGGElement>) => {
   379→   379→      if (!dragging) return;
   380→   380→      try {
   381→   381→        (e.currentTarget as Element).releasePointerCapture(e.pointerId);
   382→   382→      } catch {
   383→   383→        /* pointer already released */
   384→   384→      }
   385→   385→      setDragging(null);
   386→   386→    },
   387→   387→    [dragging],
   388→   388→  );
   389→   389→
   390→   390→  const onKeyDownControl = useCallback(
   391→   391→    (e: React.KeyboardEvent<SVGGElement>, target: DragTarget) => {
   392→   392→      const step = e.shiftKey ? 0.1 : 0.01;
   393→   393→      let handled = true;
   394→   394→      if (target === "p1") {
   395→   395→        if (e.key === "ArrowLeft")
   396→   396→          setVals((p) => ({ ...p, x1: clamp(p.x1 - step, 0, 1) }));
   397→   397→        else if (e.key === "ArrowRight")
   398→   398→          setVals((p) => ({ ...p, x1: clamp(p.x1 + step, 0, 1) }));
   399→   399→        else if (e.key === "ArrowUp")
   400→   400→          setVals((p) => ({ ...p, y1: clamp(p.y1 + step, Y_MIN, Y_MAX) }));
   401→   401→        else if (e.key === "ArrowDown")
   402→   402→          setVals((p) => ({ ...p, y1: clamp(p.y1 - step, Y_MIN, Y_MAX) }));
   403→   403→        else handled = false;
   404→   404→      } else {
   405→   405→        if (e.key === "ArrowLeft")
   406→   406→          setVals((p) => ({ ...p, x2: clamp(p.x2 - step, 0, 1) }));
   407→   407→        else if (e.key === "ArrowRight")
   408→   408→          setVals((p) => ({ ...p, x2: clamp(p.x2 + step, 0, 1) }));
   409→   409→        else if (e.key === "ArrowUp")
   410→   410→          setVals((p) => ({ ...p, y2: clamp(p.y2 + step, Y_MIN, Y_MAX) }));
   411→   411→        else if (e.key === "ArrowDown")
   412→   412→          setVals((p) => ({ ...p, y2: clamp(p.y2 - step, Y_MIN, Y_MAX) }));
   413→   413→        else handled = false;
   414→   414→      }
   415→   415→      if (handled) e.preventDefault();
   416→   416→    },
   417→   417→    [],
   418→   418→  );
   419→   419→
   420→   420→  const isPresetActive = useCallback(
   421→   421→    (p: Preset): boolean =>
   422→   422→      Math.abs(p.x1 - x1) < 0.005 &&
   423→   423→      Math.abs(p.y1 - y1) < 0.005 &&
   424→   424→      Math.abs(p.x2 - x2) < 0.005 &&
   425→   425→      Math.abs(p.y2 - y2) < 0.005,
   426→   426→    [x1, y1, x2, y2],
   427→   427→  );
   428→   428→
   429→   429→  /* ── Pre-computed SVG coordinates ─────────────────────────────────── */
   430→   430→  const p0 = { x: 0, y: VB };
   431→   431→  const p1Svg = { x: x1 * VB, y: (1 - y1) * VB };
   432→   432→  const p2Svg = { x: x2 * VB, y: (1 - y2) * VB };
   433→   433→  const p3 = { x: VB, y: 0 };

   103→/* ─── scrollToSection utility ────────────────────────────────
   104→   Navigating to sections BELOW the effects grid is tricky because
   105→   VirtualScrollGrid lazy-loads cards, shifting the document height
   106→   during smooth scroll. This function:
   107→   1. If the target is below the effects grid, dispatches a "load all
   108→      cards" event to stabilize the height.
   109→   2. Waits two animation frames for React to flush + DOM to render.
   110→   3. Then smooth-scrolls to the target.
   111→   For targets above the grid, it scrolls directly (no height shift). */
   112→function scrollToSection(id: string) {
   113→  const target = document.querySelector(id) as HTMLElement | null;
   114→  if (!target) return;
   115→  target.scrollIntoView({ behavior: "smooth", block: "start" });
   116→}

   218→/* ─── Install Command ───────────────────────────────────────── */
   219→function InstallCommand() {
   220→  const [copied, setCopied] = useState(false);
   221→  const [clicked, setClicked] = useState(false);
   222→  const handleCopy = async () => {
   223→    try {
   224→      await navigator.clipboard.writeText("npm install roycss");
   225→      setCopied(true);
   226→      setClicked(true);
   227→      setTimeout(() => setCopied(false), 2000);
   228→      setTimeout(() => setClicked(false), 2500);
   229→    } catch {
   230→      /* noop */
   231→    }
   232→  };
   233→
   234→  return (
   235→    <MagneticButton
   236→      strength={0.25}
   237→      className="inline-block"
   238→    >
   239→      <div
   240→        role="button"
   241→        tabIndex={0}
   242→        onKeyDown={(e) => {
   243→          if (e.key === "Enter" || e.key === " ") {
   244→            e.preventDefault();
   245→            handleCopy();
   246→          }
   247→        }}
   248→        className={`inline-flex items-center gap-2 rounded-xl px-4 py-2.5 group cursor-pointer transition-all outline-none border-2 ${
   249→          copied
   250→            ? "border-emerald-500 bg-emerald-500/10 shadow-lg shadow-emerald-500/20"
   251→            : clicked
   252→            ? "border-primary bg-primary/10 shadow-lg shadow-primary/20"
   253→            : "border-border/60 glass-strong hover:border-primary/50 hover:shadow-md focus-visible:border-primary focus-visible:ring-2 focus-visible:ring-primary/30"
   254→        }`}
   255→        onClick={handleCopy}
   256→        aria-label="Copy npm install roycss command"
   257→        aria-pressed={copied}
   258→      >
   259→        <span className={`text-sm font-mono transition-colors ${copied ? "text-emerald-600 dark:text-emerald-400" : "text-muted-foreground"}`}>$</span>
   260→        <code className={`text-sm font-mono transition-colors ${copied ? "text-emerald-600 dark:text-emerald-400" : "text-foreground"}`}>npm install roycss</code>

   542→/* ─── 3D Tilt Stage (hero logo wrapper) ─────────────────────── */
   543→function TiltStage({ children }: { children: React.ReactNode }) {
   544→  const stageRef = useRef<HTMLDivElement>(null);
   545→
   546→  const handleMouseMove = (e: React.MouseEvent<HTMLDivElement>) => {
   547→    const el = stageRef.current;
   548→    if (!el) return;
   549→    const rect = el.getBoundingClientRect();
   550→    // Normalize pointer position to [-0.5, 0.5] around center
   551→    const px = (e.clientX - rect.left) / rect.width - 0.5;
   552→    const py = (e.clientY - rect.top) / rect.height - 0.5;
   553→    // Rotate up to ±15deg; invert Y so the logo "leans into" the cursor
   554→    const maxTilt = 15;
   555→    const tiltY = px * maxTilt * 2;
   556→    const tiltX = -py * maxTilt * 2;
   557→    el.style.setProperty("--tilt-x", `${tiltX.toFixed(2)}deg`);
   558→    el.style.setProperty("--tilt-y", `${tiltY.toFixed(2)}deg`);
   559→  };
   560→
   561→  const handleMouseLeave = () => {
   562→    const el = stageRef.current;
   563→    if (!el) return;
   564→    el.style.setProperty("--tilt-x", "0deg");
   565→    el.style.setProperty("--tilt-y", "0deg");

   580→/* ─── Featured Carousel — rotates through ALL effects ──────── */
   581→const FEATURED_BATCH_SIZE = 4;
   582→const FEATURED_INTERVAL_MS = 6000; // 6s per batch → full cycle ≈ 19 min for 1569+ effects
   583→
   584→/* useSyncExternalStore helpers for prefers-reduced-motion.
   585→   This is the React-idiomatic way to read an external system (the OS
   586→   accessibility setting) without setState-in-useEffect. */
   587→function subscribeReducedMotion(callback: () => void) {
   588→  const mq = window.matchMedia("(prefers-reduced-motion: reduce)");
   589→  mq.addEventListener("change", callback);
   590→  return () => mq.removeEventListener("change", callback);
   591→}
   592→function getReducedMotionSnapshot() {
   593→  return window.matchMedia("(prefers-reduced-motion: reduce)").matches;
   594→}
   595→function getReducedMotionSSR() {
   596→  return false; // Server: assume no reduced motion
   597→}

   599→function FeaturedCarousel({ onSelectEffect }: { onSelectEffect: (effect: CSSEffect) => void }) {
   600→  const [batchIndex, setBatchIndex] = useState(0);
   601→  // userPaused: null = user hasn't toggled → follow prefers-reduced-motion;
   602→  //             true/false = user explicitly paused or played.
   603→  const [userPaused, setUserPaused] = useState<boolean | null>(null);
   604→  const [isHovered, setIsHovered] = useState(false);
   605→
   606→  const prefersReducedMotion = useSyncExternalStore(
   607→    subscribeReducedMotion,
   608→    getReducedMotionSnapshot,
   609→    getReducedMotionSSR,
   610→  );
   611→
   612→  // Derive isPaused: user's explicit choice wins, otherwise follow OS setting.
   613→  const isPaused = userPaused ?? prefersReducedMotion;
   614→
   615→  const totalBatches = Math.ceil(effects.length / FEATURED_BATCH_SIZE);
   616→
   617→  const currentBatch = useMemo(() => {
   618→    const start = batchIndex * FEATURED_BATCH_SIZE;
   619→    return effects.slice(start, start + FEATURED_BATCH_SIZE);
   620→  }, [batchIndex]);
   621→
   622→  // Inject CSS for previous (exit anim) + current + next (preload) batches.
   623→  // This keeps the <style> tag small (~12 effects ≈ 12 KB) while ensuring
   624→  // seamless transitions and no FOUC when the batch advances.
   625→  const cssToInject = useMemo(() => {
   626→    const ids = new Set<string>();
   627→    for (let offset = -FEATURED_BATCH_SIZE; offset < FEATURED_BATCH_SIZE * 2; offset++) {
   628→      const raw = batchIndex * FEATURED_BATCH_SIZE + offset;
   629→      const idx = ((raw % effects.length) + effects.length) % effects.length;
   630→      ids.add(effects[idx].id);
   631→    }
   632→    return effects.filter((e) => ids.has(e.id)).map((e) => e.cssCode).join("\n\n");
   633→  }, [batchIndex]);
   634→
   635→  const goToPrev = useCallback(() => {
   636→    setBatchIndex((prev) => (prev - 1 + totalBatches) % totalBatches);
   637→  }, [totalBatches]);
   638→  const goToNext = useCallback(() => {
   639→    setBatchIndex((prev) => (prev + 1) % totalBatches);
   640→  }, [totalBatches]);
   641→
   642→  const startIdx = batchIndex * FEATURED_BATCH_SIZE + 1;
   643→  const endIdx = Math.min((batchIndex + 1) * FEATURED_BATCH_SIZE, effects.length);
   644→  const progressPaused = isPaused || isHovered;
   645→
   646→  return (
   647→    <section
   648→      className="py-16 sm:py-20 relative overflow-hidden z-10"
   649→      aria-label="Featured effects carousel"
   650→    >
   651→      <div className="absolute inset-0 -z-10 bg-grid opacity-20 roycss-fade-mask-b" />
   652→
   653→      {/* Scoped CSS for the effects currently on stage (and neighbours) */}
   654→      <style dangerouslySetInnerHTML={{ __html: cssToInject }} />
   655→      {/* Keyframes for the progress bar (doubles as the auto-advance timer) */}
   656→      <style>{`@keyframes roy-featured-progress { from { width: 0% } to { width: 100% } }`}</style>
   657→
   658→      <div className="container mx-auto px-4 sm:px-6">
   659→        <SectionHeading
   660→          eyebrow="Rotating showcase"
   661→          title="Featured Effects"
   662→          subtitle={`Every effect gets its moment in the spotlight — cycling through all ${effects.length} in an infinite loop.`}
   663→        />
   664→
   665→        {/* Controls + progress */}
   666→        <div
   667→          className="mt-8 flex items-center justify-between gap-3 flex-wrap"
   668→          role="toolbar"
   669→          aria-label="Featured carousel controls"
   670→        >
   671→          {/* Prev / counter / Next */}
   672→          <div className="flex items-center gap-2">
   673→            <button
   674→              onClick={goToPrev}
   675→              aria-label="Previous batch of effects"
   676→              className="flex items-center justify-center size-9 rounded-lg bg-muted/80 border border-border/50 text-muted-foreground hover:text-foreground hover:bg-muted transition-all cursor-pointer"
   677→            >
   678→              <ChevronLeft className="size-4" />
   679→            </button>
   680→            <span className="text-xs font-mono text-muted-foreground tabular-nums whitespace-nowrap px-1">
   681→              {startIdx}–{endIdx} <span className="opacity-50">/</span> {effects.length}
   682→            </span>
   683→            <button
   684→              onClick={goToNext}
   685→              aria-label="Next batch of effects"
   686→              className="flex items-center justify-center size-9 rounded-lg bg-muted/80 border border-border/50 text-muted-foreground hover:text-foreground hover:bg-muted transition-all cursor-pointer"
   687→            >
   688→              <ChevronRight className="size-4" />
   689→            </button>
   690→          </div>
   691→
   692→          {/* Progress bar — its CSS animation IS the auto-advance timer.
   693→              onAnimationEnd fires exactly when the bar fills, which advances
   694→              the batch. Pausing (button or hover) freezes both bar + timer. */}
   695→          <div
   696→            className="flex-1 min-w-[6rem] max-w-xs h-1.5 bg-muted rounded-full overflow-hidden"
   697→            aria-hidden="true"
   698→          >
   699→            <div
   700→              key={batchIndex}
   701→              className="h-full bg-primary rounded-full"
   702→              style={{
   703→                animation: `roy-featured-progress ${FEATURED_INTERVAL_MS}ms linear forwards`,
   704→                animationPlayState: progressPaused ? "paused" : "running",
   705→              }}
   706→              onAnimationEnd={() => {
   707→                if (!progressPaused) goToNext();
   708→              }}
   709→            />
   710→          </div>
   711→
   712→          {/* Play/Pause + loop badge */}
   713→          <div className="flex items-center gap-3">
   714→            <button
   715→              onClick={() => setUserPaused(!isPaused)}
   716→              aria-label={isPaused ? "Play carousel" : "Pause carousel"}
   717→              aria-pressed={isPaused}
   718→              className="flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-muted/80 border border-border/50 text-xs font-medium text-muted-foreground hover:text-foreground hover:bg-muted transition-all cursor-pointer"
   719→            >
   720→              {isPaused ? <Play className="size-3.5" /> : <Pause className="size-3.5" />}
   721→              {isPaused ? "Play" : "Pause"}
   722→            </button>
   723→            <span className="hidden sm:flex items-center gap-1 text-xs text-muted-foreground">
   724→              <Repeat className="size-3" />
   725→              Infinite loop
   726→            </span>
   727→          </div>
   728→        </div>
   729→
   730→        {/* Featured cards grid — pauses on hover so users can linger */}
   731→        <div
   732→          className="mt-8 relative z-10"
   733→          onMouseEnter={() => setIsHovered(true)}
   734→          onMouseLeave={() => setIsHovered(false)}
   735→          aria-live="polite"
   736→          aria-label={`Showing effects ${startIdx} through ${endIdx} of ${effects.length}`}
   737→        >
   738→          <AnimatePresence mode="wait">
   739→            <div key={batchIndex} className="grid grid-cols-1 md:grid-cols-2 gap-5">
   740→              {currentBatch.map((effect, i) => (
   741→                <motion.div
   742→                  key={effect.id}
   743→                  initial={{ opacity: 0, y: 16 }}
   744→                  animate={{ opacity: 1, y: 0 }}
   745→                  exit={{ opacity: 0, y: -16 }}
   746→                  transition={{ duration: 0.4, delay: i * 0.06, ease: [0.22, 1, 0.36, 1] }}
   747→                >
   748→                  <FeaturedCard effect={effect} onSelect={onSelectEffect} />
   749→                </motion.div>
   750→              ))}
   751→            </div>
   752→          </AnimatePresence>
   753→        </div>
   754→      </div>
   755→    </section>
   756→  );
   757→}

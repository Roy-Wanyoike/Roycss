   130→/* ─── Theme Toggle ──────────────────────────────────────────── */
   131→function ThemeToggle() {
   132→  const [dark, setDark] = useState(true);
   133→
   134→  useEffect(() => {
   135→    const root = document.documentElement;
   136→    if (dark) {
   137→      root.classList.add("dark");
   138→    } else {
   139→      root.classList.remove("dark");
   140→    }
   141→  }, [dark]);
   142→
   143→  return (
   144→    <button
   145→      onClick={() => setDark(!dark)}
   146→      className="flex items-center justify-center size-11 rounded-xl glass text-muted-foreground hover:text-foreground transition-all hover:-translate-y-0.5 cursor-pointer"
   147→      aria-label="Toggle theme"
   148→    >
   149→      {dark ? <Sun className="size-4" /> : <Moon className="size-4" />}

   430→/* ─── 3D Tilt Stage (hero logo wrapper) ─────────────────────── */
   431→function TiltStage({ children }: { children: React.ReactNode }) {
   432→  const stageRef = useRef<HTMLDivElement>(null);
   433→
   434→  const handleMouseMove = (e: React.MouseEvent<HTMLDivElement>) => {
   435→    const el = stageRef.current;
   436→    if (!el) return;
   437→    const rect = el.getBoundingClientRect();
   438→    // Normalize pointer position to [-0.5, 0.5] around center
   439→    const px = (e.clientX - rect.left) / rect.width - 0.5;
   440→    const py = (e.clientY - rect.top) / rect.height - 0.5;
   441→    // Rotate up to ±15deg; invert Y so the logo "leans into" the cursor
   442→    const maxTilt = 15;
   443→    const tiltY = px * maxTilt * 2;
   444→    const tiltX = -py * maxTilt * 2;
   445→    el.style.setProperty("--tilt-x", `${tiltX.toFixed(2)}deg`);
   446→    el.style.setProperty("--tilt-y", `${tiltY.toFixed(2)}deg`);
   447→  };
   448→
   449→  const handleMouseLeave = () => {
   450→    const el = stageRef.current;
   451→    if (!el) return;
   452→    el.style.setProperty("--tilt-x", "0deg");
   453→    el.style.setProperty("--tilt-y", "0deg");
   454→  };
   455→
   456→  return (
   457→    <div
   458→      ref={stageRef}
   459→      className="roycss-tilt-stage inline-block"
   460→      onMouseMove={handleMouseMove}
   461→      onMouseLeave={handleMouseLeave}
   462→    >
   463→      <div className="roycss-tilt-target">{children}</div>
   464→    </div>
   465→  );
   466→}

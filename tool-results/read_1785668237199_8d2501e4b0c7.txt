    17→/**
    18→ * EasingVisualizer — an interactive cubic-bezier easing curve designer.
    19→ *
    20→ * Features:
    21→ *  - SVG curve editor with two draggable control points (pointer + keyboard).
    22→ *  - Live moving dot driven by a `requestAnimationFrame` loop.
    23→ *  - Bidirectionally synced numeric inputs + sliders (x1, y1, x2, y2).
    24→ *  - Preset gallery (linear / ease / quad / cubic / back / bounce-ish) with
    25→ *    mini curve thumbnails; active preset is highlighted.
    26→ *  - Real-world live preview: a 48px circle traversing a track using the
    27→ *    actual `cubic-bezier(...)` CSS transition.
    28→ *  - Generated `transition-timing-function` CSS with copy-to-clipboard.
    29→ *  - Configurable preview duration (100–5000ms).
    30→ *
    31→ * Math:
    32→ *  - The cubic-bezier parametric form with P0=(0,0), P1=(x1,y1),
    33→ *    P2=(x2,y2), P3=(1,1):
    34→ *      B(t) = 3(1-t)^2 t P1 + 3(1-t) t^2 P2 + t^3 P3   (P0 term is 0)
    35→ *  - For the moving dot we need y given x (time). Because x1,x2 ∈ [0,1],
    36→ *    Bx(t) is monotonically increasing in t, so a 40-iteration binary
    37→ *    search on t converges to sub-pixel precision.
    38→ */
    39→
    40→interface BezierVals {
    41→  x1: number;
    42→  y1: number;
    43→  x2: number;
    44→  y2: number;
    45→}

    79→function clamp(n: number, lo: number, hi: number): number {
    80→  return Math.min(hi, Math.max(lo, n));
    81→}
    82→
    83→/** Trim trailing zeros so 0.5 stays "0.5" and 0.42 stays "0.42". */
    84→function fmt(n: number): string {
    85→  return Number(n.toFixed(3)).toString();

    88→/**
    89→ * Solve `y = By(t)` for a given `x = Bx(t)` using binary search.
    90→ * Requires x1, x2 ∈ [0,1] (guaranteed by clamping) so Bx is monotonic.
    91→ */
    92→function bezierYForX(
    93→  x: number,
    94→  x1: number,
    95→  y1: number,
    96→  x2: number,
    97→  y2: number,
    98→): number {
    99→  let lo = 0;
   100→  let hi = 1;
   101→  for (let i = 0; i < 40; i++) {
   102→    const t = (lo + hi) / 2;
   103→    const mt = 1 - t;
   104→    const bx = 3 * mt * mt * t * x1 + 3 * mt * t * t * x2 + t * t * t;
   105→    if (bx < x) lo = t;
   106→    else hi = t;
   107→  }
   108→  const t = (lo + hi) / 2;
   109→  const mt = 1 - t;
   110→  return 3 * mt * mt * t * y1 + 3 * mt * t * t * y2 + t * t * t;
   111→}
   112→
   113→/** Sample the bezier into a polyline SVG path (robust, no curve command needed). */
   114→function sampleCurvePath(
   115→  x1: number,
   116→  y1: number,
   117→  x2: number,
   118→  y2: number,
   119→  samples: number,
   120→  scale: number,
   121→): string {
   122→  const pts: string[] = [];
   123→  for (let i = 0; i <= samples; i++) {
   124→    const t = i / samples;
   125→    const mt = 1 - t;
   126→    const x = 3 * mt * mt * t * x1 + 3 * mt * t * t * x2 + t * t * t;
   127→    const y = 3 * mt * mt * t * y1 + 3 * mt * t * t * y2 + t * t * t;
   128→    pts.push(`${(x * scale).toFixed(2)},${((1 - y) * scale).toFixed(2)}`);
   129→  }
   130→  return `M ${pts.join(" L ")}`;
   131→}

   136→    () => sampleCurvePath(x1, y1, x2, y2, 20, 24),

   151→        strokeWidth="1.5"
   152→        strokeLinecap="round"
   153→        strokeLinejoin="round"
   154→      />
   155→    </svg>
   156→  );
   157→}

   225→export function EasingVisualizer() {
   226→  const [vals, setVals] = useState<BezierVals>(DEFAULT_VALS);
   227→  const [duration, setDuration] = useState(1500);
   228→  const [playing, setPlaying] = useState(true);
   229→  const [progress, setProgress] = useState(0);
   230→  const [previewPos, setPreviewPos] = useState<0 | 1>(0);
   231→  const [copied, setCopied] = useState(false);
   232→  const [dragging, setDragging] = useState<DragTarget | null>(null);
   233→
   234→  const svgRef = useRef<SVGSVGElement | null>(null);
   235→  const trackRef = useRef<HTMLDivElement | null>(null);
   236→  const [trackWidth, setTrackWidth] = useState(0);
   237→
   238→  const rafRef = useRef<number | null>(null);
   239→  const lastTimeRef = useRef(0);
   240→
   241→  const { x1, y1, x2, y2 } = vals;
   242→
   243→  /* ── Track width (for live preview translate distance) ─────────────── */
   244→  useEffect(() => {
   245→    const el = trackRef.current;
   246→    if (!el) return;
   247→    const update = () => setTrackWidth(el.offsetWidth);
   248→    update();
   249→    const ro = new ResizeObserver(update);
   250→    ro.observe(el);
   251→    return () => ro.disconnect();
   252→  }, []);
   253→
   254→  /* ── Moving dot: requestAnimationFrame loop, cleaned up on unmount ── */
   255→  useEffect(() => {
   256→    if (!playing) {
   257→      lastTimeRef.current = 0;
   258→      return;
   259→    }
   260→    const tick = (now: number) => {
   261→      if (lastTimeRef.current === 0) lastTimeRef.current = now;
   262→      const dt = now - lastTimeRef.current;
   263→      lastTimeRef.current = now;
   264→      setProgress((p) => {
   265→        const next = p + dt / duration;
   266→        // Loop seamlessly, preserving sub-frame precision.
   267→        return next >= 1 ? next - Math.floor(next) : next;
   268→      });
   269→      rafRef.current = requestAnimationFrame(tick);
   270→    };
   271→    rafRef.current = requestAnimationFrame(tick);
   272→    return () => {
   273→      if (rafRef.current !== null) cancelAnimationFrame(rafRef.current);
   274→      lastTimeRef.current = 0;
   275→    };
   276→  }, [playing, duration]);
   277→
   278→  /* ── Live preview box: CSS transition toggled on an interval ──────── */
   279→  useEffect(() => {
   280→    if (!playing) return;
   281→    // Kick off the back-and-forth cycle on the next animation frame so we
   282→    // don't trigger a synchronous re-render inside this effect body.
   283→    const kickoff = requestAnimationFrame(() => setPreviewPos(1));
   284→    const interval = window.setInterval(
   285→      () => setPreviewPos((p) => (p === 1 ? 0 : 1)),
   286→      duration + 120,
   287→    );
   288→    return () => {
   289→      cancelAnimationFrame(kickoff);
   290→      window.clearInterval(interval);
   291→    };
   292→  }, [playing, duration]);
   293→
   294→  const curvePath = useMemo(
   295→    () => sampleCurvePath(x1, y1, x2, y2, 60, VB),
   296→    [x1, y1, x2, y2],
   297→  );
   298→
   299→  const easedY = useMemo(
   300→    () => bezierYForX(progress, x1, y1, x2, y2),
   301→    [progress, x1, y1, x2, y2],
   302→  );
   303→
   304→  const cssString = `transition-timing-function: cubic-bezier(${fmt(x1)}, ${fmt(y1)}, ${fmt(x2)}, ${fmt(y2)});`;
   305→  const cubicBezierString = `cubic-bezier(${fmt(x1)}, ${fmt(y1)}, ${fmt(x2)}, ${fmt(y2)})`;
   306→
   307→  const handleCopy = useCallback(async () => {
   308→    try {
   309→      await navigator.clipboard.writeText(cssString);
   310→      setCopied(true);
   311→      window.setTimeout(() => setCopied(false), 2000);
   312→    } catch {
   313→      /* clipboard unavailable — silently ignore */
   314→    }
   315→  }, [cssString]);
   316→
   317→  const setVal = useCallback(
   318→    (key: keyof BezierVals, v: number) => {
   319→      setVals((prev) => ({ ...prev, [key]: v }));
   320→    },
   321→    [],
   322→  );
   323→
   324→  const handleReset = useCallback(() => {
   325→    setVals(DEFAULT_VALS);
   326→    setProgress(0);
   327→  }, []);
   328→
   329→  /* ── Pointer → normalized (x, y) conversion using the SVG CTM ─────── */
   330→  const getNormalized = useCallback(
   331→    (e: React.PointerEvent): { x: number; y: number } | null => {
   332→      const svg = svgRef.current;
   333→      if (!svg) return null;
   334→      const ctm = svg.getScreenCTM();
   335→      if (!ctm) return null;
   336→      const pt = new DOMPoint(e.clientX, e.clientY);
   337→      const svgP = pt.matrixTransform(ctm.inverse());
   338→      return {
   339→        x: clamp(svgP.x / VB, 0, 1),
   340→        y: clamp(1 - svgP.y / VB, Y_MIN, Y_MAX),
   341→      };
   342→    },
   343→    [],
   344→  );
   345→
   346→  const onPointerDownControl = useCallback(
   347→    (e: React.PointerEvent<SVGGElement>, target: DragTarget) => {
   348→      e.preventDefault();
   349→      e.stopPropagation();
   350→      (e.currentTarget as Element).setPointerCapture(e.pointerId);
   351→      setDragging(target);
   352→      const n = getNormalized(e);
   353→      if (!n) return;
   354→      if (target === "p1") {
   355→        setVals((prev) => ({ ...prev, x1: n.x, y1: n.y }));
   356→      } else {
   357→        setVals((prev) => ({ ...prev, x2: n.x, y2: n.y }));
   358→      }
   359→    },
   360→    [getNormalized],
   361→  );
   362→
   363→  const onPointerMoveControl = useCallback(
   364→    (e: React.PointerEvent<SVGGElement>) => {
   365→      if (!dragging) return;
   366→      const n = getNormalized(e);
   367→      if (!n) return;
   368→      if (dragging === "p1") {
   369→        setVals((prev) => ({ ...prev, x1: n.x, y1: n.y }));
   370→      } else {
   371→        setVals((prev) => ({ ...prev, x2: n.x, y2: n.y }));
   372→      }
   373→    },
   374→    [dragging, getNormalized],
   375→  );
   376→
   377→  const onPointerUpControl = useCallback(
   378→    (e: React.PointerEvent<SVGGElement>) => {
   379→      if (!dragging) return;
   380→      try {
   381→        (e.currentTarget as Element).releasePointerCapture(e.pointerId);
   382→      } catch {
   383→        /* pointer already released */
   384→      }
   385→      setDragging(null);
   386→    },
   387→    [dragging],
   388→  );
   389→
   390→  const onKeyDownControl = useCallback(
   391→    (e: React.KeyboardEvent<SVGGElement>, target: DragTarget) => {
   392→      const step = e.shiftKey ? 0.1 : 0.01;
   393→      let handled = true;
   394→      if (target === "p1") {
   395→        if (e.key === "ArrowLeft")
   396→          setVals((p) => ({ ...p, x1: clamp(p.x1 - step, 0, 1) }));
   397→        else if (e.key === "ArrowRight")
   398→          setVals((p) => ({ ...p, x1: clamp(p.x1 + step, 0, 1) }));
   399→        else if (e.key === "ArrowUp")
   400→          setVals((p) => ({ ...p, y1: clamp(p.y1 + step, Y_MIN, Y_MAX) }));
   401→        else if (e.key === "ArrowDown")
   402→          setVals((p) => ({ ...p, y1: clamp(p.y1 - step, Y_MIN, Y_MAX) }));
   403→        else handled = false;
   404→      } else {
   405→        if (e.key === "ArrowLeft")
   406→          setVals((p) => ({ ...p, x2: clamp(p.x2 - step, 0, 1) }));
   407→        else if (e.key === "ArrowRight")
   408→          setVals((p) => ({ ...p, x2: clamp(p.x2 + step, 0, 1) }));
   409→        else if (e.key === "ArrowUp")
   410→          setVals((p) => ({ ...p, y2: clamp(p.y2 + step, Y_MIN, Y_MAX) }));
   411→        else if (e.key === "ArrowDown")
   412→          setVals((p) => ({ ...p, y2: clamp(p.y2 - step, Y_MIN, Y_MAX) }));
   413→        else handled = false;
   414→      }
   415→      if (handled) e.preventDefault();
   416→    },
   417→    [],
   418→  );
   419→
   420→  const isPresetActive = useCallback(
   421→    (p: Preset): boolean =>
   422→      Math.abs(p.x1 - x1) < 0.005 &&
   423→      Math.abs(p.y1 - y1) < 0.005 &&
   424→      Math.abs(p.x2 - x2) < 0.005 &&
   425→      Math.abs(p.y2 - y2) < 0.005,
   426→    [x1, y1, x2, y2],
   427→  );
   428→
   429→  /* ── Pre-computed SVG coordinates ─────────────────────────────────── */
   430→  const p0 = { x: 0, y: VB };
   431→  const p1Svg = { x: x1 * VB, y: (1 - y1) * VB };
   432→  const p2Svg = { x: x2 * VB, y: (1 - y2) * VB };
   433→  const p3 = { x: VB, y: 0 };
   434→  const dotX = progress * VB;
   435→  const dotY = (1 - easedY) * VB;
   436→  const circleTranslate = previewPos === 1 ? Math.max(0, trackWidth - 48) : 0;
   437→
   438→  const INPUTS: {
   439→    key: keyof BezierVals;
   440→    label: string;
   441→    min: number;
   442→    max: number;
   443→  }[] = [
   444→    { key: "x1", label: "x1", min: 0, max: 1 },
   445→    { key: "y1", label: "y1", min: Y_MIN, max: Y_MAX },
   446→    { key: "x2", label: "x2", min: 0, max: 1 },
   447→    { key: "y2", label: "y2", min: Y_MIN, max: Y_MAX },
   448→  ];
   449→
   450→  return (
   451→    <div className="mx-auto max-w-2xl space-y-4">
   452→      {/* ── Header ──────────────────────────────────────────────────── */}
   453→      <div className="flex items-center justify-between gap-2">
   454→        <div className="flex items-center gap-2">
   455→          <Activity className="size-5 text-primary" />
   456→          <div>
   457→            <h3 className="font-semibold leading-tight">Easing Visualizer</h3>
   458→            <p className="text-xs text-muted-foreground">
   459→              Design and compare cubic-bezier easing curves
   460→            </p>
   461→          </div>
   462→        </div>
   463→        <button
   464→          type="button"
   465→          onClick={handleReset}
   466→          className="flex cursor-pointer items-center gap-1 rounded-lg bg-muted px-2.5 py-1.5 text-xs font-medium text-muted-foreground transition-colors hover:bg-muted/70 hover:text-foreground"
   467→          title="Reset to ease-in-out"
   468→        >
   469→          <RotateCcw className="size-3.5" />
   470→          Reset
   471→        </button>
   472→      </div>
   473→
   474→      <div className="grid gap-4 md:grid-cols-2">
   475→        {/* ═══ LEFT: SVG editor + numeric controls + live preview ═══ */}
   476→        <div className="space-y-4">
   477→          {/* SVG curve editor */}
   478→          <div className="rounded-xl border border-border bg-card p-4">
   479→            <div className="mb-2 flex items-center justify-between">
   480→              <span className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   481→                Curve
   482→              </span>
   483→              <button
   484→                type="button"
   485→                onClick={() => setPlaying((p) => !p)}
   486→                className="flex cursor-pointer items-center gap-1 rounded-lg bg-primary/10 px-2.5 py-1 text-xs font-medium text-primary transition-colors hover:bg-primary/20"
   487→                title={playing ? "Pause animation" : "Play animation"}
   488→              >
   489→                {playing ? (
   490→                  <Pause className="size-3.5" />
   491→                ) : (
   492→                  <Play className="size-3.5" />
   493→                )}
   494→                {playing ? "Pause" : "Play"}
   495→              </button>
   496→            </div>
   497→
   498→            <div className="relative py-4">
   499→              <svg
   500→                ref={svgRef}
   501→                viewBox={`0 0 ${VB} ${VB}`}
   502→                className="mx-auto h-auto w-full max-w-[300px] touch-none select-none overflow-visible"
   503→                role="img"
   504→                aria-label="Cubic bezier curve editor"
   505→              >
   506→                {/* Grid lines at 0.25 intervals */}
   507→                {[0, 0.25, 0.5, 0.75, 1].map((v) => (
   508→                  <g key={v}>
   509→                    <line
   510→                      x1={v * VB}
   511→                      y1={0}
   512→                      x2={v * VB}
   513→                      y2={VB}
   514→                      className="stroke-border"
   515→                      strokeWidth={1}
   516→                      strokeDasharray="2 4"
   517→                    />

    23→/**
    24→ * SpecificityCalculator — a self-contained CSS specificity scoring tool.
    25→ *
    26→ * Paste a list of CSS selectors (one per line, or comma-separated) and the tool
    27→ * computes the standard (a, b, c) specificity tuple for each:
    28→ *   - a = ID selectors              (#id)
    29→ *   - b = class / attribute / pseudo-class selectors  (.x, [x], :hover)
    30→ *   - c = type / pseudo-element selectors             (div, ::before)
    31→ *
    32→ * Special functional pseudo-classes:
    33→ *   - :where(...)  contributes 0 to the score
    34→ *   - :is(...) / :not(...) contribute the MAX specificity of their inner selectors
    35→ *
    36→ * The integer score `a*100 + b*10 + c` is shown for ranking readability, but
    37→ * the TRUE sort order uses tuple comparison (the integer form breaks down once
    38→ * any component exceeds 9 — e.g. 11 classes ties with 1 id + 1 class).
    39→ *
    40→ * Limitations (v1):
    41→ *   - Commas inside :not() / :is() / :where() are handled correctly because
    42→ *     the splitter tracks paren/bracket depth; commas inside attribute
    43→ *     selectors `[a="x,y"]` are also handled. However, an unescaped top-level
    44→ *     comma is always treated as a selector-list separator.
    45→ *   - Namespaced type selectors (`svg|circle`) are counted as a single type.
    46→ *   - Legacy single-colon pseudo-elements (`:before`, `:after`, `:first-line`,
    47→ *     `:first-letter`) are recognised as type selectors (c), per CSS spec.
    48→ */
    49→
    50→interface Specificity {
    51→  a: number;
    52→  b: number;
    53→  c: number;
    54→}
    55→
    56→interface ParsedSelector {
    57→  /** Original (trimmed) selector text. */
    58→  selector: string;
    59→  /** Computed specificity tuple. */
    60→  spec: Specificity;
    61→  /** `a*100 + b*10 + c` — readability only, NOT used for true ranking. */
    62→  score: number;
    63→}
    64→
    65→const LEGACY_PSEUDO_ELEMENTS = new Set([
    66→  "before",
    67→  "after",
    68→  "first-line",
    69→  "first-letter",
    70→]);
    71→
    72→/** Strip /* ... *​/ comments (including multiline). */
    73→function stripComments(input: string): string {
    74→  return input.replace(/\/\*[\s\S]*?\*\//g, "");
    75→}
    76→
    77→/** Test for an identifier character: [A-Za-z0-9_-] plus non-ASCII. */
    78→function isIdentChar(ch: string): boolean {
    79→  return /[\w-]/.test(ch);
    80→}
    81→
    82→/**
    83→ * Read the contents of a balanced parenthesised group, starting at `s[i]`
    84→ * which MUST be '('. Returns the inner substring (excluding the outer parens).
    85→ * Does NOT mutate `i` — use `skipParens` for that.
    86→ */
    87→function readParens(s: string, i: number): string {
    88→  if (s[i] !== "(") return "";
    89→  let depth = 0;
    90→  let j = i;
    91→  const n = s.length;
    92→  while (j < n) {
    93→    const ch = s[j];
    94→    if (ch === "(") depth++;
    95→    else if (ch === ")") {
    96→      depth--;
    97→      if (depth === 0) return s.slice(i + 1, j);
    98→    } else if (ch === '"' || ch === "'") {
    99→      const quote = ch;
   100→      j++;
   101→      while (j < n && s[j] !== quote) j++;
   102→    }
   103→    j++;
   104→  }
   105→  return s.slice(i + 1); // unbalanced — return what we have
   106→}
   107→
   108→/**
   109→ * Return the index immediately after the closing ')' that matches the '(' at
   110→ * `s[i]`. If unbalanced, returns `s.length`.
   111→ */
   112→function skipParens(s: string, i: number): number {
   113→  if (s[i] !== "(") return i;
   114→  let depth = 0;
   115→  let j = i;
   116→  const n = s.length;
   117→  while (j < n) {
   118→    const ch = s[j];
   119→    if (ch === "(") depth++;
   120→    else if (ch === ")") {
   121→      depth--;
   122→      if (depth === 0) return j + 1;
   123→    } else if (ch === '"' || ch === "'") {
   124→      const quote = ch;
   125→      j++;
   126→      while (j < n && s[j] !== quote) j++;
   127→    }
   128→    j++;
   129→  }
   130→  return n;
   131→}
   132→
   133→/** Compare two specificity tuples. Returns >0 if x>y, <0 if x<y, 0 if equal. */
   134→function compareSpec(x: Specificity, y: Specificity): number {
   135→  if (x.a !== y.a) return x.a - y.a;
   136→  if (x.b !== y.b) return x.b - y.b;
   137→  return x.c - y.c;
   138→}
   139→
   140→/**
   141→ * Split a selector list on top-level commas — commas inside (), [], or quoted
   142→ * strings are preserved. Used both for the user input and for the inner
   143→ * arguments of :is() / :not() / :where().
   144→ */
   145→function splitTopLevelCommas(input: string): string[] {
   146→  const out: string[] = [];
   147→  let depth = 0;
   148→  let current = "";
   149→  let i = 0;
   150→  const n = input.length;
   151→  while (i < n) {
   152→    const ch = input[i];
   153→    if (ch === '"' || ch === "'") {
   154→      const quote = ch;
   155→      current += ch;
   156→      i++;
   157→      while (i < n) {
   158→        current += input[i];
   159→        if (input[i] === quote) {
   160→          i++;
   161→          break;
   162→        }
   163→        i++;
   164→      }
   165→      continue;
   166→    }
   167→    if (ch === "(" || ch === "[") {
   168→      depth++;
   169→      current += ch;
   170→    } else if (ch === ")" || ch === "]") {
   171→      depth = Math.max(0, depth - 1);
   172→      current += ch;
   173→    } else if (ch === "," && depth === 0) {
   174→      out.push(current);
   175→      current = "";
   176→    } else {
   177→      current += ch;
   178→    }
   179→    i++;
   180→  }
   181→  out.push(current);
   182→  return out.map((s) => s.trim()).filter((s) => s.length > 0);
   183→}

   185→/** Core recursive specificity computation for a single selector string. */
   186→function computeSpecificity(input: string): Specificity {
   187→  const s = stripComments(input).trim();
   188→  let a = 0;
   189→  let b = 0;
   190→  let c = 0;
   191→  let i = 0;
   192→  const n = s.length;
   193→
   194→  while (i < n) {
   195→    const ch = s[i];
   196→
   197→    if (ch === "#") {
   198→      a++;
   199→      i++;
   200→      while (i < n && isIdentChar(s[i])) i++;
   201→    } else if (ch === ".") {
   202→      b++;
   203→      i++;
   204→      while (i < n && isIdentChar(s[i])) i++;
   205→    } else if (ch === "[") {
   206→      b++;
   207→      i++;
   208→      // Skip until matching ], honouring quoted strings.
   209→      while (i < n && s[i] !== "]") {
   210→        if (s[i] === '"' || s[i] === "'") {
   211→          const quote = s[i];
   212→          i++;
   213→          while (i < n && s[i] !== quote) i++;
   214→          if (i < n) i++;
   215→        } else {
   216→          i++;
   217→        }
   218→      }
   219→      if (i < n) i++; // consume ']'
   220→    } else if (ch === ":") {
   221→      if (s[i + 1] === ":") {
   222→        // pseudo-element (::name)
   223→        c++;
   224→        i += 2;
   225→        while (i < n && isIdentChar(s[i])) i++;
   226→        if (i < n && s[i] === "(") i = skipParens(s, i);
   227→      } else {
   228→        // pseudo-class (:name or :name(args))
   229→        i++;
   230→        let name = "";
   231→        while (i < n && isIdentChar(s[i])) {
   232→          name += s[i];
   233→          i++;
   234→        }
   235→        if (i < n && s[i] === "(") {
   236→          const inner = readParens(s, i);
   237→          const after = skipParens(s, i);
   238→          const lowerName = name.toLowerCase();
   239→          if (lowerName === "where") {
   240→            // :where() contributes 0 to all components — do nothing.
   241→          } else if (
   242→            lowerName === "is" ||
   243→            lowerName === "not" ||
   244→            lowerName === "matches"
   245→          ) {
   246→            // contribute MAX specificity of inner selector list
   247→            const innerList = splitTopLevelCommas(inner);
   248→            let maxA = 0;
   249→            let maxB = 0;
   250→            let maxC = 0;
   251→            for (const innerSel of innerList) {
   252→              const sub = computeSpecificity(innerSel);
   253→              if (compareSpec(sub, { a: maxA, b: maxB, c: maxC }) > 0) {
   254→                maxA = sub.a;
   255→                maxB = sub.b;
   256→                maxC = sub.c;
   257→              }
   258→            }
   259→            a += maxA;
   260→            b += maxB;
   261→            c += maxC;
   262→          } else {
   263→            // other functional pseudo-class (:nth-child, :has, :dir, ...)
   264→            // — counts as a single pseudo-class (b). The arguments do not
   265→            // add specificity (with the rare exception of :nth-child(an+b of S)
   266→            // which is intentionally not modelled in v1).
   267→            b++;
   268→          }
   269→          i = after;
   270→        } else {
   271→          // bare pseudo-class — but watch for legacy pseudo-elements
   272→          if (LEGACY_PSEUDO_ELEMENTS.has(name.toLowerCase())) {
   273→            c++;
   274→          } else {
   275→            b++;
   276→          }
   277→        }
   278→      }
   279→    } else if (ch === "*") {
   280→      // universal selector — contributes nothing
   281→      i++;
   282→    } else if (
   283→      ch === " " ||
   284→      ch === "\t" ||
   285→      ch === "\n" ||
   286→      ch === "\r" ||
   287→      ch === "+" ||
   288→      ch === ">" ||
   289→      ch === "~"
   290→    ) {
   291→      // combinators — ignore
   292→      i++;
   293→    } else if (ch === "|" && s[i + 1] === "|") {
   294→      // column combinator || — ignore
   295→      i += 2;
   296→    } else if (isIdentChar(ch) || ch === "\\") {
   297→      // type selector (possibly namespace-prefixed: foo|bar counts as ONE type)
   298→      c++;
   299→      while (i < n && isIdentChar(s[i])) i++;
   300→      // skip namespaced suffix: |bar
   301→      if (i < n && s[i] === "|" && s[i + 1] !== "|") {
   302→        i++;
   303→        while (i < n && isIdentChar(s[i])) i++;
   304→      }
   305→    } else {
   306→      // any other stray character — ignore
   307→      i++;
   308→    }
   309→  }
   310→
   311→  return { a, b, c };
   312→}

   314→/**
   315→ * Parse a textarea blob into an array of ParsedSelector records. Selectors may
   316→ * be separated by newlines and/or top-level commas. Returns the count of
   317→ * skipped (empty or whitespace-only) fragments for UI feedback.
   318→ */
   319→function parseSelectors(raw: string): {
   320→  results: ParsedSelector[];
   321→  skipped: number;
   322→} {
   323→  const cleaned = stripComments(raw);
   324→  // First split by newlines, then by top-level commas on each line.
   325→  const lines = cleaned.split(/\r?\n/);
   326→  const fragments: string[] = [];
   327→  for (const line of lines) {
   328→    for (const piece of splitTopLevelCommas(line)) {
   329→      fragments.push(piece);
   330→    }
   331→  }
   332→
   333→  const results: ParsedSelector[] = [];
   334→  let skipped = 0;
   335→  for (const frag of fragments) {
   336→    const selector = frag.trim();
   337→    if (selector.length === 0) {
   338→      skipped++;
   339→      continue;
   340→    }
   341→    const spec = computeSpecificity(selector);
   342→    const score = spec.a * 100 + spec.b * 10 + spec.c;
   343→    results.push({ selector, spec, score });
   344→  }
   345→  return { results, skipped };
   346→}

   372→export function SpecificityCalculator() {
   373→  const [input, setInput] = useState("");
   374→  const [sortAsc, setSortAsc] = useState(false);
   375→  const [copied, setCopied] = useState(false);
   376→  const [howOpen, setHowOpen] = useState(false);
   377→
   378→  const { results, skipped } = useMemo(() => parseSelectors(input), [input]);
   379→
   380→  const sorted = useMemo(() => {
   381→    const copy = [...results];
   382→    copy.sort((x, y) =>
   383→      sortAsc ? compareSpec(x.spec, y.spec) : compareSpec(y.spec, x.spec)
   384→    );
   385→    return copy;
   386→  }, [results, sortAsc]);
   387→
   388→  const highest = useMemo(() => {
   389→    if (results.length === 0) return null;
   390→    return results.reduce((max, r) =>
   391→      compareSpec(r.spec, max.spec) > 0 ? r : max
   392→    );
   393→  }, [results]);
   394→
   395→  const lowest = useMemo(() => {
   396→    if (results.length === 0) return null;
   397→    return results.reduce((min, r) =>
   398→      compareSpec(r.spec, min.spec) < 0 ? r : min
   399→    );
   400→  }, [results]);
   401→
   402→  const maxScore = useMemo(() => {
   403→    if (sorted.length === 0) return 1;
   404→    return Math.max(1, ...sorted.map((r) => r.score));
   405→  }, [sorted]);
   406→
   407→  const handleLoadExample = useCallback(() => {
   408→    setInput(EXAMPLE_SELECTORS);
   409→  }, []);
   410→
   411→  const handleClear = useCallback(() => {
   412→    setInput("");
   413→    setCopied(false);
   414→  }, []);
   415→
   416→  const handleCopy = useCallback(async () => {
   417→    if (sorted.length === 0) return;
   418→    const lines = [
   419→      `# CSS Specificity Report`,
   420→      `Selectors analyzed: ${sorted.length}`,
   421→      highest ? `Highest: ${specToTuple(highest.spec)}` : "",
   422→      lowest ? `Lowest:  ${specToTuple(lowest.spec)}` : "",
   423→      "",
   424→      ...sorted.map(
   425→        (r) =>
   426→          `${specToTuple(r.spec).padEnd(12)} ${String(r.score).padStart(4)}  ${r.selector}`
   427→      ),
   428→    ].filter((l) => l.length >= 0);
   429→    const text = lines.join("\n");
   430→    try {
   431→      await navigator.clipboard.writeText(text);
   432→      setCopied(true);
   433→      window.setTimeout(() => setCopied(false), 2000);
   434→    } catch {
   435→      /* clipboard unavailable — silent */
   436→    }
   437→  }, [sorted, highest, lowest]);
   438→
   439→  return (
   440→    <div className="w-full max-w-2xl mx-auto space-y-4">
   441→      {/* Header */}
   442→      <div className="flex items-start gap-3">
   443→        <div className="grid size-10 shrink-0 place-items-center rounded-xl bg-primary/10 text-primary border border-primary/20">
   444→          <Calculator className="size-5" />
   445→        </div>
   446→        <div className="min-w-0">
   447→          <h2 className="text-lg font-semibold leading-tight text-foreground">
   448→            CSS Specificity Calculator
   449→          </h2>
   450→          <p className="text-sm text-muted-foreground mt-0.5">
   451→            Paste selectors (one per line, or comma-separated) and rank them by specificity.
   452→          </p>
   453→        </div>
   454→      </div>
   455→
   456→      {/* Input */}
   457→      <div className="space-y-2">
   458→        <label
   459→          htmlFor="specificity-input"
   460→          className="text-xs font-semibold text-muted-foreground uppercase tracking-wider"
   461→        >
   462→          Selectors
   463→        </label>
   464→        <Textarea
   465→          id="specificity-input"
   466→          value={input}
   467→          onChange={(e) => setInput(e.target.value)}
   468→          placeholder={"#nav .item\ndiv.card:hover\nul li a.active"}
   469→          rows={5}
   470→          spellCheck={false}
   471→          className="font-mono text-sm leading-relaxed resize-y scrollbar-thin"
   472→          aria-describedby="specificity-help"
   473→        />
   474→        <p id="specificity-help" className="text-[11px] text-muted-foreground">
   475→          One selector per line, or comma-separated. Commas inside{" "}
   476→          <code className="font-mono">:not()</code>,{" "}
   477→          <code className="font-mono">:is()</code> and attribute values are
   478→          preserved.
   479→        </p>
   480→      </div>
   481→
   482→      {/* Actions */}
   483→      <div className="flex flex-wrap items-center gap-2">
   484→        <Button size="sm" variant="default" onClick={handleLoadExample}>
   485→          <Sparkles className="size-3.5" />
   486→          Load example
   487→        </Button>
   488→        <Button
   489→          size="sm"
   490→          variant="outline"
   491→          onClick={handleClear}
   492→          disabled={input.length === 0}

   494→          <Trash2 className="size-3.5" />

   496→        </Button>

   498→          size="sm"
   499→          variant="outline"
   500→          onClick={handleCopy}

   502→          aria-label="Copy results summary to clipboard"

   507→            <Copy className="size-3.5" />

   512→          size="sm"

   514→          onClick={() => setSortAsc((v) => !v)}

   526→      {/* Summary */}

   529→          <span className="text-muted-foreground">
   530→            <span className="font-semibold text-foreground">{sorted.length}</span>{" "}
   531→            selector{sorted.length === 1 ? "" : "s"} analyzed
   532→          </span>
   533→          <span className="text-muted-foreground/40" aria-hidden>·</span>
   534→          {highest && (
   535→            <span className="text-muted-foreground">
   536→              Highest:{" "}
   537→              <Badge variant="secondary" className="ml-1 font-mono">
   538→                {specToTuple(highest.spec)}
   539→              </Badge>
   540→            </span>
   541→          )}
   542→          <span className="text-muted-foreground/40" aria-hidden>·</span>
   543→          {lowest && (
   544→            <span className="text-muted-foreground">
   545→              Lowest:{" "}
   546→              <Badge variant="outline" className="ml-1 font-mono">
   547→                {specToTuple(lowest.spec)}
   548→              </Badge>
   549→            </span>
   550→          )}
   551→          {skipped > 0 && (
   552→            <>
   553→              <span className="text-muted-foreground/40" aria-hidden>·</span>

   559→        </div>

   570→              exit={{ opacity: 0 }}

   579→              <p className="text-xs text-muted-foreground max-w-xs">

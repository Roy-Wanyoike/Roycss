    59→
    60→const PLAYGROUND_EXAMPLES = [
    61→  "A glassmorphism card that glows on hover",
    62→  "A neon text effect with flicker animation",
    63→  "A loader with 3 bouncing dots",
    64→  "A button with a shine sweep on hover",
    65→  "An aurora gradient background",

    67→
    68→function AIPlayground() {
    69→  const [prompt, setPrompt] = useState("");
    70→  const [css, setCss] = useState("");
    71→  const [loading, setLoading] = useState(false);
    72→  const [error, setError] = useState("");
    73→  const [copied, setCopied] = useState(false);
    74→  const [replayKey, setReplayKey] = useState(0);
    75→
    76→  const handleGenerate = useCallback(async () => {
    77→    if (!prompt.trim() || loading) return;
    78→    setLoading(true);
    79→    setError("");
    80→    setCss("");
    81→    try {
    82→      const res = await fetch("/api/ai-playground", {
    83→        method: "POST",
    84→        headers: { "Content-Type": "application/json" },
    85→        body: JSON.stringify({ prompt }),
    86→      });
    87→      const data = await res.json();
    88→      if (!res.ok) throw new Error(data.error || "Request failed");
    89→      setCss(data.css);
    90→      setReplayKey((k) => k + 1);
    91→    } catch (e) {
    92→      setError(e instanceof Error ? e.message : "Failed to generate");
    93→    } finally {
    94→      setLoading(false);
    95→    }
    96→  }, [prompt, loading]);
    97→
    98→  const handleCopy = useCallback(async () => {
    99→    try {
   100→      await navigator.clipboard.writeText(css);
   101→      setCopied(true);
   102→      setTimeout(() => setCopied(false), 2000);
   103→    } catch { /* noop */ }
   104→  }, [css]);
   105→
   106→  // Extract the first class name for preview
   107→  const previewClass = useMemo(() => {
   108→    const match = css.match(/\.roycss-([a-z0-9-]+)/);
   109→    return match ? `roycss-${match[1]}` : null;
   110→  }, [css]);
   111→
   112→  return (
   113→    <div className="space-y-4">
   114→      <div>
   115→        <label className="text-xs font-semibold text-muted-foreground uppercase tracking-wider mb-2 block">
   116→          Describe the effect you want
   117→        </label>
   118→        <div className="flex gap-2">
   119→          <input
   120→            type="text"
   121→            value={prompt}
   122→            onChange={(e) => setPrompt(e.target.value)}
   123→            onKeyDown={(e) => e.key === "Enter" && handleGenerate()}
   124→            placeholder="e.g., A pulsing neon border for a card"
   125→            className="flex-1 h-11 px-4 rounded-xl bg-background border border-border/50 focus:border-primary/50 text-sm text-foreground focus:outline-none transition-all"
   126→            disabled={loading}
   127→          />
   128→          <button
   129→            onClick={handleGenerate}
   130→            disabled={loading || !prompt.trim()}
   131→            className="flex items-center gap-1.5 px-5 h-11 rounded-xl bg-primary text-primary-foreground text-sm font-medium hover:bg-primary/90 disabled:opacity-50 disabled:cursor-not-allowed transition-all cursor-pointer shrink-0"
   132→          >
   133→            {loading ? <Loader2 className="size-4 animate-spin" /> : <Sparkles className="size-4" />}
   134→            {loading ? "Generating..." : "Generate"}
   135→          </button>
   136→        </div>
   137→      </div>
   138→
   139→      {/* Example prompts */}
   140→      <div className="flex items-center gap-1.5 flex-wrap">
   141→        <span className="text-xs text-muted-foreground mr-1">Try:</span>
   142→        {PLAYGROUND_EXAMPLES.slice(0, 3).map((ex) => (
   143→          <button
   144→            key={ex}
   145→            onClick={() => setPrompt(ex)}
   146→            disabled={loading}
   147→            className="px-2.5 py-1 rounded-lg text-xs bg-muted/60 text-muted-foreground hover:bg-muted hover:text-foreground transition-all cursor-pointer disabled:opacity-50"
   148→          >
   149→            {ex.length > 35 ? ex.substring(0, 33) + "…" : ex}
   150→          </button>
   151→        ))}
   152→      </div>
   153→
   154→      {error && (
   155→        <div className="flex items-center gap-2 p-3 rounded-xl bg-rose-500/10 text-rose-500 text-sm">
   156→          <AlertCircle className="size-4 shrink-0" />
   157→          {error}
   158→        </div>
   159→      )}
   160→
   161→      <AnimatePresence>
   162→        {css && (
   163→          <motion.div
   164→            initial={{ opacity: 0, y: 10 }}
   165→            animate={{ opacity: 1, y: 0 }}
   166→            className="space-y-3"
   167→          >
   168→            {/* Live preview */}
   169→            <div className="rounded-xl border border-border/60 overflow-hidden">
   170→              <div className="flex items-center justify-between px-3 py-2 border-b border-border/40 bg-muted/20">
   171→                <span className="text-xs font-semibold text-muted-foreground uppercase tracking-wider">Live Preview</span>
   172→                <button
   173→                  onClick={() => setReplayKey((k) => k + 1)}
   174→                  className="flex items-center gap-1 text-xs text-primary hover:text-primary/80 cursor-pointer"
   175→                >
   176→                  <RefreshCw className="size-3" /> Replay
   177→                </button>
   178→              </div>
   179→              <div
   180→                key={replayKey}
   181→                className="h-40 flex items-center justify-center"
   182→                style={{ background: "oklch(0.15 0.02 250)" }}
   183→              >
   184→                {previewClass && (
   185→                  <style dangerouslySetInnerHTML={{ __html: css }} />
   186→                )}
   187→                {previewClass && (
   188→                  <div className={previewClass} style={{ width: "60px", height: "60px" }} />
   189→                )}
   190→              </div>
   191→            </div>
   192→
   193→            {/* Generated CSS */}
   194→            <div className="rounded-xl border border-border/60 overflow-hidden">
   195→              <div className="flex items-center justify-between px-3 py-2 border-b border-border/40 bg-muted/20">
   196→                <span className="text-xs font-semibold text-muted-foreground uppercase tracking-wider">Generated CSS</span>
   197→                <button
   198→                  onClick={handleCopy}
   199→                  className={`flex items-center gap-1 px-2.5 py-1 rounded-lg text-xs font-medium transition-all cursor-pointer ${
   200→                    copied ? "bg-emerald-500/15 text-emerald-500" : "bg-primary/10 text-primary hover:bg-primary/20"
   201→                  }`}
   202→                >
   203→                  {copied ? <Check className="size-3" /> : <Copy className="size-3" />}
   204→                  {copied ? "Copied!" : "Copy CSS"}
   205→                </button>
   206→              </div>
   207→              <pre className="p-3 overflow-x-auto scrollbar-thin max-h-64 overflow-y-auto text-xs leading-relaxed font-mono text-foreground/80">
   208→                <code>{css}</code>
   209→              </pre>
   210→            </div>
   211→          </motion.div>
   212→        )}
   213→      </AnimatePresence>
   214→    </div>
   215→  );
   216→}

   222→interface CSSIssue {
   223→  severity: "critical" | "warning" | "info";
   224→  category: string;
   225→  line: number;
   226→  message: string;
   227→  fix: string;

   229→
   230→function CSSDoctor() {
   231→  const [css, setCss] = useState("");
   232→  const [result, setResult] = useState<{ score: number; issues: CSSIssue[]; summary: string } | null>(null);
   233→  const [loading, setLoading] = useState(false);
   234→  const [error, setError] = useState("");
   235→
   236→  const handleAnalyze = useCallback(async () => {
   237→    if (!css.trim() || loading) return;
   238→    setLoading(true);
   239→    setError("");
   240→    setResult(null);
   241→    try {
   242→      const res = await fetch("/api/css-doctor", {
   243→        method: "POST",
   244→        headers: { "Content-Type": "application/json" },
   245→        body: JSON.stringify({ css }),
   246→      });
   247→      const data = await res.json();
   248→      if (!res.ok) throw new Error(data.error || "Analysis failed");
   249→      setResult(data);
   250→    } catch (e) {
   251→      setError(e instanceof Error ? e.message : "Failed to analyze");
   252→    } finally {
   253→      setLoading(false);
   254→    }
   255→  }, [css, loading]);
   256→
   257→  const scoreColor = result
   258→    ? result.score >= 80 ? "text-emerald-500" : result.score >= 50 ? "text-amber-500" : "text-rose-500"
   259→    : "";
   260→
   261→  return (
   262→    <div className="space-y-4">
   263→      <div>
   264→        <label className="text-xs font-semibold text-muted-foreground uppercase tracking-wider mb-2 block">
   265→          Paste your CSS to diagnose
   266→        </label>
   267→        <textarea
   268→          value={css}
   269→          onChange={(e) => setCss(e.target.value)}
   270→          placeholder=".my-element {\n  background: #10b981;\n  margin-left: 10px;\n  animation: pulse 2s infinite;\n}"
   271→          className="w-full h-40 p-3 rounded-xl bg-background border border-border/50 focus:border-primary/50 text-xs font-mono text-foreground focus:outline-none transition-all resize-none scrollbar-thin"
   272→          disabled={loading}
   273→        />
   274→      </div>
   275→
   276→      <button
   277→        onClick={handleAnalyze}
   278→        disabled={loading || !css.trim()}
   279→        className="flex items-center gap-1.5 px-5 h-11 rounded-xl bg-primary text-primary-foreground text-sm font-medium hover:bg-primary/90 disabled:opacity-50 disabled:cursor-not-allowed transition-all cursor-pointer"
   280→      >
   281→        {loading ? <Loader2 className="size-4 animate-spin" /> : <Stethoscope className="size-4" />}
   282→        {loading ? "Analyzing..." : "Run Diagnosis"}
   283→      </button>
   284→
   285→      {error && (
   286→        <div className="flex items-center gap-2 p-3 rounded-xl bg-rose-500/10 text-rose-500 text-sm">
   287→          <AlertCircle className="size-4 shrink-0" />
   288→          {error}
   289→        </div>
   290→      )}
   291→
   292→      <AnimatePresence>
   293→        {result && (
   294→          <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} className="space-y-3">
   295→            {/* Score */}
   296→            <div className="flex items-center justify-between p-4 rounded-xl bg-muted/30 border border-border/50">
   297→              <div>
   298→                <p className="text-xs font-semibold text-muted-foreground uppercase tracking-wider">Health Score</p>
   299→                <p className={`font-display text-3xl font-bold ${scoreColor}`}>{result.score}/100</p>
   300→              </div>
   301→              <p className="text-sm text-muted-foreground max-w-xs text-right">{result.summary}</p>
   302→            </div>
   303→
   304→            {/* Issues */}
   305→            <div className="space-y-2">
   306→              <p className="text-xs font-semibold text-muted-foreground uppercase tracking-wider">
   307→                {result.issues.length} {result.issues.length === 1 ? "issue" : "issues"} found
   308→              </p>
   309→              {result.issues.map((issue, i) => {
   310→                const Icon = issue.severity === "critical" ? AlertCircle : issue.severity === "warning" ? AlertTriangle : Info;
   311→                const color = issue.severity === "critical" ? "text-rose-500" : issue.severity === "warning" ? "text-amber-500" : "text-sky-500";
   312→                const bg = issue.severity === "critical" ? "bg-rose-500/5" : issue.severity === "warning" ? "bg-amber-500/5" : "bg-sky-500/5";
   313→                return (
   314→                  <div key={i} className={`p-3 rounded-xl ${bg} border border-border/50`}>
   315→                    <div className="flex items-start gap-2 mb-1.5">
   316→                      <Icon className={`size-4 ${color} shrink-0 mt-0.5`} />
   317→                      <div className="min-w-0 flex-1">
   318→                        <div className="flex items-center gap-2 mb-0.5">
   319→                          <Badge variant="secondary" className="text-xs px-1.5 py-0 capitalize">{issue.category}</Badge>
   320→                          {issue.line > 0 && <span className="text-xs text-muted-foreground">Line {issue.line}</span>}
   321→                        </div>
   322→                        <p className="text-sm text-foreground">{issue.message}</p>
   323→                      </div>
   324→                    </div>
   325→                    {issue.fix && (
   326→                      <pre className="mt-2 p-2 rounded-lg bg-background/60 text-xs font-mono text-emerald-600 dark:text-emerald-400 overflow-x-auto scrollbar-thin">
   327→                        <code>{issue.fix}</code>
   328→                      </pre>
   329→                    )}
   330→                  </div>
   331→                );
   332→              })}
   333→            </div>
   334→          </motion.div>
   335→        )}
   336→      </AnimatePresence>
   337→    </div>
   338→  );
   339→}

   341→/* ═══════════════════════════════════════════════════════════════
   342→   3. Utility Explorer — hover any class, see its CSS + perf
   343→   ═══════════════════════════════════════════════════════════════ */
   344→
   345→import { effects } from "@/lib/roycss-effects";
   346→
   347→function UtilityExplorer() {
   348→  const [search, setSearch] = useState("");
   349→  const [selected, setSelected] = useState<CSSEffect | null>(effects[0]);
   350→
   351→  const filtered = useMemo(() => {
   352→    const q = search.toLowerCase().trim();
   353→    if (!q) return effects.slice(0, 50);
   354→    return effects.filter(e => e.name.toLowerCase().includes(q) || e.id.includes(q) || e.tags.some(t => t.includes(q))).slice(0, 50);
   355→  }, [search]);
   356→
   357→  // Compute "performance metrics" from the CSS
   358→  const metrics = useMemo(() => {
   359→    if (!selected) return null;
   360→    const css = selected.cssCode;
   361→    const lines = css.split("\n").length;
   362→    const size = new Blob([css]).size;
   363→    const hasKeyframes = /@keyframes/.test(css);
   364→    const hasPseudoElements = /::before|::after/.test(css);
   365→    const hasCustomProps = /@property/.test(css);
   366→    const animDuration = css.match(/animation:[^;]*?(\d+(?:\.\d+)?)s/)?.[1];
   367→    const usesColorMix = /color-mix/.test(css);
   368→    const usesOKLCH = /oklch/.test(css);
   369→    const usesLogicalProps = /inline-start|inline-end|block-start|block-end|inset-inline/.test(css);
   370→
   371→    return {
   372→      lines,
   373→      size,
   374→      hasKeyframes,
   375→      hasPseudoElements,
   376→      hasCustomProps,
   377→      animDuration: animDuration ? `${animDuration}s` : "none",
   378→      usesColorMix,
   379→      usesOKLCH,
   380→      usesLogicalProps,
   381→      score: [
   382→        usesOKLCH, usesColorMix, usesLogicalProps, hasKeyframes, lines < 30
   383→      ].filter(Boolean).length * 20,
   384→    };
   385→  }, [selected]);
   386→
   387→  return (
   388→    <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 h-[60vh]">
   389→      {/* Effect list */}
   390→      <div className="flex flex-col rounded-xl border border-border/60 overflow-hidden">
   391→        <div className="p-2 border-b border-border/40 bg-muted/20">
   392→          <input
   393→            type="search"
   394→            placeholder="Search effects..."
   395→            value={search}
   396→            onChange={(e) => setSearch(e.target.value)}
   397→            className="w-full h-9 px-3 rounded-lg bg-background border border-border/50 focus:border-primary/50 text-sm focus:outline-none"
   398→          />
   399→        </div>
   400→        <div className="flex-1 overflow-y-auto scrollbar-thin p-1.5">
   401→          {filtered.map((e) => (
   402→            <button
   403→              key={e.id}
   404→              onClick={() => setSelected(e)}
   405→              onMouseEnter={() => setSelected(e)}
   406→              className={`w-full flex items-center gap-2 p-2 rounded-lg transition-all cursor-pointer text-left ${
   407→                selected?.id === e.id ? "bg-primary/10" : "hover:bg-muted/50"
   408→              }`}
   409→            >
   410→              <div className="flex items-center justify-center size-9 rounded-lg bg-muted/40 border border-border/50 overflow-hidden shrink-0">
   411→                <div className="scale-[0.4] origin-center">
   412→                  <LivePreview effect={e} />
   413→                </div>
   414→              </div>
   415→              <div className="min-w-0 flex-1">
   416→                <p className="text-sm font-medium text-foreground truncate">{e.name}</p>
   417→                <p className="text-xs text-muted-foreground truncate font-mono">.roycss-{e.id}</p>
   418→              </div>
   419→            </button>
   420→          ))}
   421→        </div>
   422→      </div>
   423→
   424→      {/* Detail panel */}
   425→      <div className="flex flex-col rounded-xl border border-border/60 overflow-hidden">
   426→        {selected && metrics ? (
   427→          <>
   428→            <div className="p-3 border-b border-border/40 bg-muted/20">
   429→              <p className="text-sm font-semibold text-foreground">{selected.name}</p>
   430→              <p className="text-xs text-muted-foreground font-mono">.roycss-{selected.id}</p>
   431→            </div>
   432→            <div className="flex-1 overflow-y-auto scrollbar-thin p-3 space-y-3">
   433→              {/* Preview */}
   434→              <div className="h-24 rounded-lg flex items-center justify-center" style={{ background: "oklch(0.15 0.02 250)" }}>
   435→                <LivePreview effect={selected} />
   436→              </div>
   437→
   438→              {/* Score */}
   439→              <div className="flex items-center justify-between p-2.5 rounded-lg bg-muted/30">
   440→                <span className="text-xs font-semibold text-muted-foreground uppercase tracking-wider">Quality Score</span>
   441→                <span className={`font-display text-xl font-bold ${metrics.score >= 80 ? "text-emerald-500" : metrics.score >= 60 ? "text-amber-500" : "text-rose-500"}`}>
   442→                  {metrics.score}/100
   443→                </span>
   444→              </div>
   445→
   446→              {/* Metrics grid */}
   447→              <div className="grid grid-cols-2 gap-2">
   448→                <Metric label="Lines" value={String(metrics.lines)} />
   449→                <Metric label="Size" value={`${metrics.size}B`} />
   450→                <Metric label="Duration" value={metrics.animDuration} />
   451→                <Metric label="Keyframes" value={metrics.hasKeyframes ? "Yes" : "No"} />
   452→              </div>
   453→
   454→              {/* Compliance */}
   455→              <div className="space-y-1.5">
   456→                <p className="text-xs font-semibold text-muted-foreground uppercase tracking-wider">Standards Compliance</p>
   457→                <ComplianceRow label="OKLCH colors" passed={metrics.usesOKLCH} />
   458→                <ComplianceRow label="color-mix()" passed={metrics.usesColorMix} />
   459→                <ComplianceRow label="Logical properties" passed={metrics.usesLogicalProps} />
   460→                <ComplianceRow label="@keyframes defined" passed={metrics.hasKeyframes} />
   461→                <ComplianceRow label="@property registered" passed={metrics.hasCustomProps} />
   462→                <ComplianceRow label="Pseudo-elements" passed={metrics.hasPseudoElements} />
   463→              </div>
   464→            </div>
   465→          </>
   466→        ) : (
   467→          <div className="flex-1 flex items-center justify-center text-sm text-muted-foreground">
   468→            Hover an effect to explore
   469→          </div>
   470→        )}
   471→      </div>
   472→    </div>
   473→  );
   474→}
   475→
   476→function Metric({ label, value }: { label: string; value: string }) {
   477→  return (
   478→    <div className="p-2 rounded-lg bg-muted/30">
   479→      <p className="text-xs text-muted-foreground">{label}</p>
   480→      <p className="text-sm font-mono font-medium text-foreground">{value}</p>
   481→    </div>
   482→  );
   483→}
   484→
   485→function ComplianceRow({ label, passed }: { label: string; passed: boolean }) {
   486→  return (
   487→    <div className="flex items-center justify-between text-xs">
   488→      <span className="text-muted-foreground">{label}</span>
   489→      {passed ? (
   490→        <Check className="size-3.5 text-emerald-500" />
   491→      ) : (
   492→        <X className="size-3.5 text-muted-foreground/40" />
   493→      )}
   494→    </div>
   495→  );
   496→}

   498→/* ═══════════════════════════════════════════════════════════════
   499→   4. Benchmark — live performance test of any effect
   500→   ═══════════════════════════════════════════════════════════════ */
   501→
   502→function Benchmark() {
   503→  const [search, setSearch] = useState("");
   504→  const [selected, setSelected] = useState<CSSEffect | null>(effects[0]);
   505→  const [results, setResults] = useState<{ fps: number; renderMs: number; domNodes: number; layoutShifts: number } | null>(null);
   506→  const [running, setRunning] = useState(false);
   507→
   508→  const filtered = useMemo(() => {
   509→    const q = search.toLowerCase().trim();
   510→    if (!q) return effects.slice(0, 30);
   511→    return effects.filter(e => e.name.toLowerCase().includes(q) || e.id.includes(q)).slice(0, 30);
   512→  }, [search]);
   513→
   514→  const runBenchmark = useCallback(async () => {
   515→    if (!selected || running) return;
   516→    setRunning(true);
   517→    setResults(null);
   518→
   519→    // Create a test container with 100 instances of the effect
   520→    const container = document.createElement("div");
   521→    container.style.cssText = "position:fixed;top:-9999px;left:-9999px;width:800px;visibility:hidden;";
   522→    container.id = "roycss-bench-container";
   523→
   524→    // Inject the CSS
   525→    const style = document.createElement("style");
   526→    style.textContent = selected.cssCode;
   527→    container.appendChild(style);
   528→
   529→    // Create 100 elements
   530→    for (let i = 0; i < 100; i++) {
   531→      const el = document.createElement("div");
   532→      el.className = `roycss-${selected.id}`;
   533→      el.style.cssText = "width:40px;height:40px;margin:4px;display:inline-block;";
   534→      container.appendChild(el);
   535→    }
   536→
   537→    document.body.appendChild(container);
   538→
   539→    // Wait a frame for layout
   540→    await new Promise(r => requestAnimationFrame(r));
   541→
   542→    // Measure render time
   543→    const startRender = performance.now();
   544→    for (let i = 0; i < 10; i++) {
   545→      container.style.display = "none";
   546→      void container.offsetHeight; // force reflow
   547→      container.style.display = "";
   548→      void container.offsetHeight; // force reflow
   549→    }
   550→    const renderMs = (performance.now() - startRender) / 10;
   551→
   552→    // Measure FPS over 1 second
   553→    const domNodes = container.querySelectorAll("*").length;
   554→    let frames = 0;
   555→    const start = performance.now();
   556→    return new Promise<void>((resolve) => {
   557→      const measure = () => {
   558→        frames++;
   559→        if (performance.now() - start < 1000) {
   560→          requestAnimationFrame(measure);
   561→        } else {
   562→          const fps = Math.round(frames);
   563→          // Layout shifts (approximate — check if elements moved)
   564→          const firstEl = container.children[1] as HTMLElement;
   565→          const rect1 = firstEl.getBoundingClientRect();
   566→          container.style.width = "600px";
   567→          const rect2 = firstEl.getBoundingClientRect();
   568→          const layoutShifts = rect1.left !== rect2.left ? 1 : 0;
   569→          container.style.width = "800px";
   570→
   571→          setResults({ fps, renderMs: Math.round(renderMs * 100) / 100, domNodes, layoutShifts });
   572→          document.body.removeChild(container);
   573→          setRunning(false);
   574→          resolve();
   575→        }
   576→      };
   577→      requestAnimationFrame(measure);
   578→    });
   579→  }, [selected, running]);
   580→
   581→  return (
   582→    <div className="space-y-4">
   583→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   584→        {/* Effect selector */}
   585→        <div>
   586→          <label className="text-xs font-semibold text-muted-foreground uppercase tracking-wider mb-2 block">
   587→            Select effect to benchmark
   588→          </label>
   589→          <input
   590→            type="search"
   591→            placeholder="Search effects..."
   592→            value={search}
   593→            onChange={(e) => setSearch(e.target.value)}
   594→            className="w-full h-9 px-3 mb-2 rounded-lg bg-background border border-border/50 focus:border-primary/50 text-sm focus:outline-none"
   595→          />
   596→          <div className="max-h-48 overflow-y-auto scrollbar-thin space-y-0.5">
   597→            {filtered.map((e) => (
   598→              <button
   599→                key={e.id}
   600→                onClick={() => setSelected(e)}
   601→                className={`w-full flex items-center gap-2 p-1.5 rounded-lg transition-all cursor-pointer text-left ${
   602→                  selected?.id === e.id ? "bg-primary/10" : "hover:bg-muted/50"
   603→                }`}
   604→              >
   605→                <div className="flex items-center justify-center size-7 rounded bg-muted/40 border border-border/50 overflow-hidden shrink-0">
   606→                  <div className="scale-[0.35] origin-center">
   607→                    <LivePreview effect={e} />
   608→                  </div>
   609→                </div>
   610→                <span className="text-xs font-medium text-foreground truncate">{e.name}</span>
   611→              </button>
   612→            ))}
   613→          </div>
   614→        </div>
   615→
   616→        {/* Results */}
   617→        <div className="space-y-3">
   618→          <button
   619→            onClick={runBenchmark}
   620→            disabled={running || !selected}
   621→            className="w-full flex items-center justify-center gap-1.5 px-5 h-11 rounded-xl bg-primary text-primary-foreground text-sm font-medium hover:bg-primary/90 disabled:opacity-50 disabled:cursor-not-allowed transition-all cursor-pointer"
   622→          >
   623→            {running ? <Loader2 className="size-4 animate-spin" /> : <Play className="size-4" />}
   624→            {running ? "Benchmarking (100 elements, 1s)..." : "Run Benchmark"}
   625→          </button>
   626→
   627→          <AnimatePresence>
   628→            {results && (
   629→              <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} className="space-y-2">
   630→                <BenchmarkMetric
   631→                  label="Frames per second"
   632→                  value={String(results.fps)}
   633→                  unit="fps"
   634→                  good={results.fps >= 55}
   635→                  ok={results.fps >= 30}
   636→                />
   637→                <BenchmarkMetric
   638→                  label="Render time (100 elements)"
   639→                  value={String(results.renderMs)}
   640→                  unit="ms"
   641→                  good={results.renderMs < 5}
   642→                  ok={results.renderMs < 20}
   643→                />
   644→                <BenchmarkMetric
   645→                  label="DOM nodes created"
   646→                  value={String(results.domNodes)}
   647→                  unit="nodes"
   648→                  good={results.domNodes < 105}
   649→                  ok={results.domNodes < 200}

   919→function AIMigration() {
   920→  const [framework, setFramework] = useState("bootstrap");
   921→  const [inputCSS, setInputCSS] = useState(MIGRATION_EXAMPLES.bootstrap);
   922→  const [outputCSS, setOutputCSS] = useState("");
   923→  const [loading, setLoading] = useState(false);
   924→  const [error, setError] = useState("");
   925→  const [copied, setCopied] = useState(false);
   926→
   927→  const handleFrameworkChange = (fw: string) => {
   928→    setFramework(fw);
   929→    setInputCSS(MIGRATION_EXAMPLES[fw] || "");
   930→    setOutputCSS("");
   931→    setError("");
   932→  };
   933→
   934→  const handleMigrate = useCallback(async () => {
   935→    if (!inputCSS.trim() || loading) return;
   936→    setLoading(true);
   937→    setError("");
   938→    setOutputCSS("");
   939→    try {
   940→      const res = await fetch("/api/ai-migration", {
   941→        method: "POST",
   942→        headers: { "Content-Type": "application/json" },
   943→        body: JSON.stringify({ css: inputCSS, framework }),
   944→      });
   945→      const data = await res.json();
   946→      if (!res.ok) throw new Error(data.error || "Migration failed");
   947→      setOutputCSS(data.css);
   948→    } catch (e) {
   949→      setError(e instanceof Error ? e.message : "Failed to migrate");
   950→    } finally {
   951→      setLoading(false);
   952→    }
   953→  }, [inputCSS, framework, loading]);
   954→
   955→  const handleCopy = useCallback(async () => {
   956→    try { await navigator.clipboard.writeText(outputCSS); setCopied(true); setTimeout(() => setCopied(false), 2000); } catch { /* noop */ }
   957→  }, [outputCSS]);
   958→
   959→  return (
   960→    <div className="space-y-4">
   961→      {/* Framework selector */}
   962→      <div>
   963→        <label className="text-xs font-semibold text-muted-foreground uppercase tracking-wider mb-2 block">Source Framework</label>
   964→        <div className="flex items-center gap-1.5 flex-wrap">
   965→          {FRAMEWORKS.map(fw => (
   966→            <button
   967→              key={fw.id}

  1249→/* ═══════════════════════════════════════════════════════════════
  1250→   8. Design Diff — before/after CSS comparison
  1251→   ═══════════════════════════════════════════════════════════════ */
  1252→
  1253→function DesignDiff() {
  1254→  const [beforeCSS, setBeforeCSS] = useState("");
  1255→  const [afterCSS, setAfterCSS] = useState("");
  1256→  const [diff, setDiff] = useState<{ added: string[]; removed: string[]; changed: { prop: string; before: string; after: string }[] } | null>(null);
  1257→
  1258→  const computeDiff = useCallback(() => {
  1259→    const parseProps = (css: string): Record<string, string> => {
  1260→      const props: Record<string, string> = {};
  1261→      const match = css.match(/\{([^}]+)\}/);
  1262→      if (!match) return props;
  1263→      for (const decl of match[1].split(";").map(d => d.trim()).filter(Boolean)) {
  1264→        const idx = decl.indexOf(":");
  1265→        if (idx > 0) props[decl.substring(0, idx).trim()] = decl.substring(idx + 1).trim();
  1266→      }
  1267→      return props;
  1268→    };
  1269→
  1270→    const before = parseProps(beforeCSS);
  1271→    const after = parseProps(afterCSS);
  1272→    const allProps = new Set([...Object.keys(before), ...Object.keys(after)]);
  1273→    const added: string[] = [];
  1274→    const removed: string[] = [];
  1275→    const changed: { prop: string; before: string; after: string }[] = [];
  1276→
  1277→    for (const prop of allProps) {
  1278→      if (!before[prop] && after[prop]) added.push(prop);
  1279→      else if (before[prop] && !after[prop]) removed.push(prop);
  1280→      else if (before[prop] !== after[prop]) changed.push({ prop, before: before[prop], after: after[prop] });
  1281→    }
  1282→
  1283→    setDiff({ added: added.sort(), removed: removed.sort(), changed: changed.sort((a, b) => a.prop.localeCompare(b.prop)) });
  1284→  }, [beforeCSS, afterCSS]);
  1285→
  1286→  return (
  1287→    <div className="space-y-4">
  1288→      <p className="text-sm text-muted-foreground">Paste two CSS blocks to see exactly what changed — properties added, removed, and modified.</p>
  1289→
  1290→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
  1291→        {/* Before */}
  1292→        <div>
  1293→          <label className="text-xs font-semibold text-muted-foreground uppercase tracking-wider mb-2 block">Before (original CSS)</label>
  1294→          <textarea
  1295→            value={beforeCSS}
  1296→            onChange={(e) => setBeforeCSS(e.target.value)}
  1297→            placeholder=".card {\n  background: #fff;\n  border-radius: 4px;\n  padding: 16px;\n}"
  1298→            className="w-full h-32 p-3 rounded-xl bg-background border border-rose-500/30 focus:border-rose-500/50 text-xs font-mono text-foreground focus:outline-none transition-all resize-none scrollbar-thin"
  1299→          />
  1300→        </div>
  1301→        {/* After */}
  1302→        <div>
  1303→          <label className="text-xs font-semibold text-muted-foreground uppercase tracking-wider mb-2 block">After (updated CSS)</label>
  1304→          <textarea
  1305→            value={afterCSS}
  1306→            onChange={(e) => setAfterCSS(e.target.value)}
  1307→            placeholder=".card {\n  background: oklch(1 0 0);\n  border-radius: 0.5rem;\n  padding: 1rem;\n  box-shadow: 0 1px 3px oklch(0 0 0 / 0.1);\n}"
  1308→            className="w-full h-32 p-3 rounded-xl bg-background border border-emerald-500/30 focus:border-emerald-500/50 text-xs font-mono text-foreground focus:outline-none transition-all resize-none scrollbar-thin"
  1309→          />
  1310→        </div>
  1311→      </div>
  1312→
  1313→      <button
  1314→        onClick={computeDiff}
  1315→        disabled={!beforeCSS.trim() || !afterCSS.trim()}
  1316→        className="flex items-center gap-1.5 px-5 h-11 rounded-xl bg-primary text-primary-foreground text-sm font-medium hover:bg-primary/90 disabled:opacity-50 disabled:cursor-not-allowed transition-all cursor-pointer"
  1317→      >
  1318→        <GitCompare className="size-4" /> Compare CSS
  1319→      </button>

  1322→        {diff && (
  1323→          <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} className="space-y-3">
  1324→            {/* Summary */}
  1325→            <div className="grid grid-cols-3 gap-2">
  1326→              <div className="p-2.5 rounded-lg bg-emerald-500/10 text-center">
  1327→                <p className="text-xs text-emerald-600 dark:text-emerald-400">Added</p>
  1328→                <p className="font-display text-xl font-bold text-emerald-500">{diff.added.length}</p>
  1329→              </div>
  1330→              <div className="p-2.5 rounded-lg bg-amber-500/10 text-center">
  1331→                <p className="text-xs text-amber-600 dark:text-amber-400">Changed</p>
  1332→                <p className="font-display text-xl font-bold text-amber-500">{diff.changed.length}</p>
  1333→              </div>

  1335→                <p className="text-xs text-rose-600 dark:text-rose-400">Removed</p>

  1340→            {/* Changed properties */}
  1341→            {diff.changed.length > 0 && (
  1342→              <div className="rounded-xl border border-border/60 overflow-hidden">
  1343→                <div className="px-3 py-2 border-b border-border/40 bg-amber-500/5">
  1344→                  <p className="text-xs font-semibold text-amber-600 dark:text-amber-400 uppercase tracking-wider">Changed Properties</p>
  1345→                </div>
  1346→                <div className="divide-y divide-border/30">

  1377→                  {diff.removed.map((p, i) => <code key={i} className="text-xs px-1.5 py-0.5 rounded bg-rose-500/10 text-rose-600 dark:text-rose-400 font-mono line-through">{p}</code>)}

   126→/* ─── Theme Toggle ──────────────────────────────────────────── */
   127→function ThemeToggle() {
   128→  const [dark, setDark] = useState(true);
   129→
   130→  useEffect(() => {
   131→    const root = document.documentElement;
   132→    if (dark) {
   133→      root.classList.add("dark");
   134→    } else {
   135→      root.classList.remove("dark");
   136→    }
   137→  }, [dark]);
   138→
   139→  return (
   140→    <button
   141→      onClick={() => setDark(!dark)}
   142→      className="flex items-center justify-center size-9 rounded-xl glass text-muted-foreground hover:text-foreground transition-all hover:-translate-y-0.5 cursor-pointer"
   143→      aria-label="Toggle theme"
   144→    >
   145→      {dark ? <Sun className="size-4" /> : <Moon className="size-4" />}
   146→    </button>
   147→  );
   148→}

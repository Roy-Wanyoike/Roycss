    28→    28→/**
    29→    29→ * FluidTypographyCalculator — a CSS `clamp()` generator for fluid typography.
    30→    30→ *
    31→    31→ * Generates a `clamp(MIN, PREFERRED, MAX)` value that scales font-size
    32→    32→ * linearly between a min and max viewport width, hitting exactly minFS at
    33→    33→ * minVW and maxFS at maxVW.
    34→    34→ *
    35→    35→ * Math derivation:
    36→    36→ *   We want `font = S + R * (1vw)` where `1vw` resolves to viewport/100 px.
    37→    37→ *   At vw = minVW (so 1vw = minVW/100 px), font = minFS:
    38→    38→ *     minFS = S + R * (minVW / 100)
    39→    39→ *   At vw = maxVW, font = maxFS:
    40→    40→ *     maxFS = S + R * (maxVW / 100)
    41→    41→ *   Subtracting:  maxFS - minFS = R * (maxVW - minVW) / 100
    42→    42→ *   =>  R = 100 * (maxFS - minFS) / (maxVW - minVW)   (vw coefficient, unitless)
    43→    43→ *   =>  S = minFS - R * (minVW / 100)                  (intercept, in font units)
    44→    44→ *
    45→    45→ *   Equivalently, let B = (maxFS - minFS) / (maxVW - minVW)  (px per px):
    46→    46→ *     R = 100 * B
    47→    47→ *     S_px = minFS - B * minVW
    48→    48→ *   For rem output, divide px values by rootFontSize (R is dimensionless).
    49→    49→ *
    50→    50→ * Live preview uses one iframe per simulated viewport — the iframe's layout
    51→    51→ * width equals the viewport width, so `vw` resolves correctly inside it.
    52→    52→ */
    53→    53→

   111→   111→function clampNum(n: number, lo: number, hi: number): number {
   112→   112→  return Math.min(hi, Math.max(lo, n));
   113→   113→}
   114→   114→
   115→   115→/** Format a number with up to `decimals` digits, trimming trailing zeros. */
   116→   116→function fmt(n: number, decimals = 2): string {
   117→   117→  return Number(n.toFixed(decimals)).toString();
   118→   118→}
   119→   119→
   120→   120→function computeClamp(
   121→   121→  minFS: number,
   122→   122→  maxFS: number,
   123→   123→  minVW: number,
   124→   124→  maxVW: number,
   125→   125→  unit: Unit,
   126→   126→  rootFontSize: number,
   127→   127→): ClampParams {
   128→   128→  // Guard against degenerate inputs (division by zero / reversed ranges).
   129→   129→  const vRange = maxVW - minVW;
   130→   130→  const safeRange = vRange === 0 ? 1 : vRange;
   131→   131→  const B = (maxFS - minFS) / safeRange;
   132→   132→  const R = 100 * B;
   133→   133→  const Spx = minFS - B * minVW;
   134→   134→
   135→   135→  if (unit === "rem") {
   136→   136→    const root = rootFontSize || 16;
   137→   137→    const minStr = `${fmt(minFS / root)}rem`;
   138→   138→    const maxStr = `${fmt(maxFS / root)}rem`;
   139→   139→    const preferredStr = `${fmt(Spx / root)}rem + ${fmt(R, 3)}vw`;
   140→   140→    return {
   141→   141→      B,
   142→   142→      R,
   143→   143→      Spx,
   144→   144→      S: Spx / root,
   145→   145→      minStr,
   146→   146→      maxStr,
   147→   147→      preferredStr,
   148→   148→      fullClamp: `clamp(${minStr}, ${preferredStr}, ${maxStr})`,
   149→   149→    };
   150→   150→  }
   151→   151→
   152→   152→  const minStr = `${fmt(minFS)}px`;
   153→   153→  const maxStr = `${fmt(maxFS)}px`;
   154→   154→  const preferredStr = `${fmt(Spx)}px + ${fmt(R, 3)}vw`;
   155→   155→  return {
   156→   156→    B,
   157→   157→    R,
   158→   158→    Spx,
   159→   159→    S: Spx,
   160→   160→    minStr,
   161→   161→    maxStr,
   162→   162→    preferredStr,
   163→   163→    fullClamp: `clamp(${minStr}, ${preferredStr}, ${maxStr})`,
   164→   164→  };
   165→   165→}

   167→   167→/** Resolved font size (px) at a given viewport width, clamping outside the ramp. */
   168→   168→function fontSizeAtV(
   169→   169→  V: number,
   170→   170→  minFS: number,
   171→   171→  maxFS: number,
   172→   172→  minVW: number,
   173→   173→  maxVW: number,
   174→   174→): number {
   175→   175→  if (V <= minVW) return minFS;
   176→   176→  if (V >= maxVW) return maxFS;
   177→   177→  const range = maxVW - minVW;
   178→   178→  if (range === 0) return minFS;
   179→   179→  return minFS + ((V - minVW) * (maxFS - minFS)) / range;
   180→   180→}

   182→   182→/** Escape HTML special characters in user-provided text. */
   183→   183→function escapeHtml(s: string): string {
   184→   184→  return s
   185→   185→    .replace(/&/g, "&amp;")
   186→   186→    .replace(/</g, "&lt;")
   187→   187→    .replace(/>/g, "&gt;")
   188→   188→    .replace(/"/g, "&quot;")
   189→   189→    .replace(/'/g, "&#39;");
   190→   190→}

   287→   287→function CurveChart({
   288→   288→  minFS,

   290→   290→  minVW,
   291→   291→  maxVW,
   292→   292→  invalid,
   293→   293→}: CurveChartProps) {
   294→   294→  const W = 280;
   295→   295→  const H = 124;
   296→   296→  const padL = 34;
   297→   297→  const padR = 12;
   298→   298→  const padT = 10;
   299→   299→  const padB = 24;
   300→   300→  const plotW = W - padL - padR;
   301→   301→  const plotH = H - padT - padB;
   302→   302→
   303→   303→  // Normalize so the chart stays sane even when inputs are reversed.
   304→   304→  const loFS = Math.min(minFS, maxFS);
   305→   305→  const hiFS = Math.max(minFS, maxFS);
   306→   306→  const loVW = Math.min(minVW, maxVW);
   307→   307→  const hiVW = Math.max(minVW, maxVW);
   308→   308→
   309→   309→  const xMax = Math.max(1600, Math.ceil((hiVW + 100) / 100) * 100);
   310→   310→  const yMaxRaw = Math.max(hiFS * 1.15, loFS + 4);
   311→   311→  const yMax = Math.max(8, Math.ceil(yMaxRaw / 4) * 4);
   312→   312→
   313→   313→  const xScale = (v: number) => padL + (v / xMax) * plotW;
   314→   314→  const yScale = (f: number) => padT + plotH - (f / yMax) * plotH;
   315→   315→
   316→   316→  const pts = [
   317→   317→    { v: 0, f: loFS },
   318→   318→    { v: loVW, f: loFS },
   319→   319→    { v: hiVW, f: hiFS },
   320→   320→    { v: xMax, f: hiFS },
   321→   321→  ];
   322→   322→  const path = pts
   323→   323→    .map(
   324→   324→      (p, i) =>
   325→   325→        `${i === 0 ? "M" : "L"} ${xScale(p.v).toFixed(2)} ${yScale(p.f).toFixed(2)}`,
   326→   326→    )
   327→   327→    .join(" ");
   328→   328→
   329→   329→  const midV = (loVW + hiVW) / 2;
   330→   330→  const midF = (loFS + hiFS) / 2;
   331→   331→
   332→   332→  const xTicks = [0, loVW, hiVW, xMax];
   333→   333→  const ariaLabel = invalid
   334→   334→    ? "Font-size curve is invalid: max font size is below min, or max viewport is not greater than min."
   335→   335→    : `Font-size curve: flat at ${fmt(loFS)}px until viewport ${loVW}px, ramps linearly to ${fmt(hiFS)}px at ${hiVW}px viewport, then flat.`;
   336→   336→
   337→   337→  return (
   338→   338→    <div className="space-y-1 rounded-xl border border-border bg-card p-4">
   339→   339→      <div className="flex items-center justify-between">
   340→   340→        <span className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   341→   341→          Font-size curve
   342→   342→        </span>
   343→   343→        <span className="font-mono text-[10px] text-muted-foreground">
   344→   344→          px ↑ / px →
   345→   345→        </span>
   346→   346→      </div>
   347→   347→      <svg
   348→   348→        viewBox={`0 0 ${W} ${H}`}
   349→   349→        className="w-full"
   350→   350→        role="img"
   351→   351→        aria-label={ariaLabel}
   352→   352→      >
   353→   353→        {/* Horizontal grid lines at loFS / hiFS */}
   354→   354→        <line
   355→   355→          x1={padL}
   356→   356→          y1={yScale(loFS)}
   357→   357→          x2={W - padR}
   358→   358→          y2={yScale(loFS)}
   359→   359→          className="stroke-muted-foreground/25"
   360→   360→          strokeWidth={0.5}
   361→   361→          strokeDasharray="2 3"
   362→   362→        />
   363→   363→        <line
   364→   364→          x1={padL}
   365→   365→          y1={yScale(hiFS)}
   366→   366→          x2={W - padR}
   367→   367→          y2={yScale(hiFS)}
   368→   368→          className="stroke-muted-foreground/25"
   369→   369→          strokeWidth={0.5}
   370→   370→          strokeDasharray="2 3"
   371→   371→        />
   372→   372→
   373→   373→        {/* Vertical guide lines at loVW / hiVW */}
   374→   374→        <line
   375→   375→          x1={xScale(loVW)}
   376→   376→          y1={padT}
   377→   377→          x2={xScale(loVW)}
   378→   378→          y2={padT + plotH}
   379→   379→          className="stroke-muted-foreground/25"
   380→   380→          strokeWidth={0.5}
   381→   381→          strokeDasharray="2 3"
   382→   382→        />
   383→   383→        <line
